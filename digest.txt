// Daftar bank & e-wallet tujuan pencairan (simulasi), satu sumber untuk backend dan web.
// Dikirim ke frontend lewat GET /api/keuangan, jadi frontend tidak perlu menyalin daftar ini.
// Ubah di sini kalau ada bank/e-wallet yang ditambah atau panjang rekeningnya berubah.

const bank = (id, nama, digit, contoh) => ({
    id,
    nama,
    digitMin: digit, // rekening bank: panjang harus tepat
    digitMax: digit,
    contoh,
});

const ewallet = (id, nama) => ({
    id,
    nama,
    digitMin: 10, // nomor HP: 10-13 digit
    digitMax: 13,
    contoh: "081234567890",
});

const DAFTAR_PENYEDIA = {
    bank: [
        bank("BCA", "Bank Central Asia (BCA)", 10, "1234567890"),
        bank("BRI", "Bank Rakyat Indonesia (BRI)", 15, "123401000123501"),
        bank("BNI", "Bank Negara Indonesia (BNI)", 10, "0123456789"),
        bank("Mandiri", "Bank Mandiri", 13, "1230004567890"),
        bank("BSI", "Bank Syariah Indonesia (BSI)", 10, "7123456789"),
        bank("CIMB", "CIMB Niaga", 13, "7012345678000"),
        bank("Danamon", "Bank Danamon", 10, "0036123456"),
        bank("Permata", "Bank Permata", 10, "4001234567"),
    ],
    eWallet: [
        ewallet("GoPay", "GoPay"),
        ewallet("OVO", "OVO"),
        ewallet("DANA", "DANA"),
        ewallet("ShopeePay", "ShopeePay"),
        ewallet("LinkAja", "LinkAja"),
    ],
};

// jenis = "Bank" | "E-Wallet"; mengembalikan null kalau id tidak ada di daftar
const cariPenyedia = (jenis, id) => {
    const daftar =
        jenis === "Bank" ? DAFTAR_PENYEDIA.bank : DAFTAR_PENYEDIA.eWallet;
    return daftar.find((p) => p.id === id) ?? null;
};

// Mengembalikan pesan error, atau null kalau nomor sudah sesuai aturan penyedia
const pesanRekening = (jenis, penyedia, angka) => {
    if (jenis === "Bank") {
        return angka.length === penyedia.digitMin
            ? null
            : `Nomor rekening ${penyedia.id} harus tepat ${penyedia.digitMin} digit angka`;
    }
    if (!angka.startsWith("08")) {
        return `Nomor HP ${penyedia.nama} harus diawali 08`;
    }
    return angka.length >= penyedia.digitMin &&
        angka.length <= penyedia.digitMax
        ? null
        : `Nomor HP ${penyedia.nama} harus ${penyedia.digitMin}-${penyedia.digitMax} digit`;
};

module.exports = { DAFTAR_PENYEDIA, cariPenyedia, pesanRekening };

const bcrypt = require("bcrypt");
const crypto = require("crypto"); // TAMBAHAN: Import modul crypto bawaan Node.js
const User = require("../models/user.model");
const { createToken } = require("../lib/jwt");

exports.login = async (req, res) => {
    try {
        const { email, password } = req.body;
        const user = await User.findByEmail(email);

        // REVISI: Pesan error disamakan agar tidak membocorkan apakah email terdaftar atau tidak
        if (!user)
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Email atau kata sandi salah",
                });

        const isValid = await bcrypt.compare(password, user.password);

        // REVISI: Pesan error disamakan
        if (!isValid)
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Email atau kata sandi salah",
                });

        const token = createToken(user);
        res.json({
            success: true,
            message: "Login berhasil",
            token,
            user: { email: user.email, role: user.role },
        });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.forgotPassword = async (req, res) => {
    try {
        const { email } = req.body;
        const user = await User.findByEmail(email);
        if (!user)
            return res
                .status(404)
                .json({ success: false, message: "Email tidak terdaftar" });

        // REVISI: Menggunakan crypto.randomInt untuk angka acak yang lebih aman
        const code = crypto.randomInt(100000, 999999).toString();

        await User.saveResetCode(email, code);

        // Simulasi pengiriman email (di proyek nyata gunakan Nodemailer)
        console.log(
            `[SIMULASI EMAIL] Kode reset untuk ${email} adalah: ${code}`,
        );

        res.json({
            success: true,
            message: "Kode reset telah dikirim (Cek terminal backend)",
        });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.verifyResetCode = async (req, res) => {
    try {
        const { email, code } = req.body;
        const user = await User.findByEmailAndResetCode(email, code);

        if (!user)
            return res.status(400).json({
                success: false,
                message: "Kode tidak valid atau kedaluwarsa",
            });
        res.json({ success: true, message: "Kode valid" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.resetPassword = async (req, res) => {
    try {
        const { email, code, newPassword } = req.body;
        const user = await User.findByEmailAndResetCode(email, code);

        if (!user)
            return res.status(400).json({
                success: false,
                message: "Kode tidak valid atau kedaluwarsa",
            });

        const hashed = await bcrypt.hash(newPassword, 10);
        await User.updatePassword(email, hashed);

        res.json({ success: true, message: "Kata sandi berhasil diperbarui" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

const Koleksi = require("../models/koleksi.model");
const {
    hapusFileGambar,
    PUBLIC_PREFIX,
} = require("../middleware/upload.middleware");

// Dipakai jika koleksi dibuat tanpa gambar (sama dengan DEFAULT_KOLEKSI_IMAGE di frontend)
const DEFAULT_GAMBAR =
    "https://images.unsplash.com/photo-1544441893-675973e31985?w=600&auto=format&fit=crop&q=80";

// Path yang disimpan di database, contoh: /uploads/koleksi/koleksi-1730000000000-ab12cd34ef56.jpg
const pathGambar = (file) => (file ? `${PUBLIC_PREFIX}${file.filename}` : null);

const isTokenKembar = (error) => error && error.code === "ER_DUP_ENTRY";

exports.getSemuaKoleksi = async (req, res) => {
    try {
        const data = await Koleksi.getAll();
        res.json({ success: true, data });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Gagal mengambil data",
        });
    }
};

exports.tambahKoleksi = async (req, res) => {
    // Field "gambar" dari body sengaja diabaikan; yang dipakai hanya file hasil upload
    const gambar = pathGambar(req.file) || DEFAULT_GAMBAR;
    try {
        const insertId = await Koleksi.create({ ...req.body, gambar });
        res.status(201).json({
            success: true,
            message: "Data koleksi berhasil ditambahkan",
            id: insertId,
        });
    } catch (error) {
        hapusFileGambar(gambar); // jangan tinggalkan file yatim jika insert gagal
        if (isTokenKembar(error)) {
            return res.status(409).json({
                success: false,
                message: "Token sudah digunakan, gunakan token lain.",
            });
        }
        res.status(500).json({
            success: false,
            message: "Gagal menambah data, pastikan token unik!",
        });
    }
};

exports.ubahKoleksi = async (req, res) => {
    const gambarBaru = pathGambar(req.file);
    try {
        const lama = await Koleksi.getById(req.params.id);
        if (!lama) {
            hapusFileGambar(gambarBaru);
            return res.status(404).json({
                success: false,
                message: "Data koleksi tidak ditemukan",
            });
        }

        // Tidak upload gambar baru -> pertahankan gambar lama
        await Koleksi.update(req.params.id, {
            ...req.body,
            gambar: gambarBaru || lama.gambar,
        });

        if (gambarBaru) hapusFileGambar(lama.gambar); // buang file lama yang sudah tergantikan
        res.json({ success: true, message: "Data berhasil diperbarui" });
    } catch (error) {
        hapusFileGambar(gambarBaru);
        if (isTokenKembar(error)) {
            return res.status(409).json({
                success: false,
                message: "Token sudah digunakan, gunakan token lain.",
            });
        }
        res.status(500).json({
            success: false,
            message: "Gagal memperbarui data",
        });
    }
};

exports.hapusKoleksi = async (req, res) => {
    try {
        const lama = await Koleksi.getById(req.params.id);
        await Koleksi.delete(req.params.id);
        if (lama) hapusFileGambar(lama.gambar);
        res.json({ success: true, message: "Data berhasil dihapus" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Gagal menghapus data",
        });
    }
};

                item.terakhir_aktif && item.detik_sejak_aktif !== null
                    ? Math.max(0, Number(item.detik_sejak_aktif))
                    : null;
            // Online hanya jika terakhir melapor Online DAN heartbeat masih baru
            const online =
                item.status === "Online" &&
                detik !== null &&
                detik <= BATAS_ONLINE_DETIK;

            return {
                id: item.id,
                name: item.nama,
                phone: item.no_telp, // null jika daftar lewat email
                email: item.email, // null jika daftar lewat telepon
                address: item.alamat,
                status: online ? "Online" : "Offline",
                inactiveSeconds: detik, // dipakai web untuk "Offline 5 menit lalu"
                createdAt: new Date(item.tanggal_dibuat).toLocaleDateString(
                    "sv-SE",
                ),
                avatar: item.avatar || DUMMY_AVATAR,
            };
        });

        res.json({ success: true, data: formattedData });
    } catch (error) {
        console.error("[USER ERROR]", error);
        res.status(500).json({
            success: false,
            message: "Gagal mengambil data pengunjung",
        });
    }
};

// Fungsi satpam untuk mengecek token
const authenticate = (req, res, next) => {
    try {
        const authHeader = req.headers.authorization;
        if (!authHeader || !authHeader.startsWith("Bearer ")) {
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Akses ditolak. Token tidak ditemukan.",
                });
        }

        const token = authHeader.split(" ")[1];
        const decoded = verifyToken(token);

        req.user = decoded; // Simpan data user (dari token) ke request
        next(); // Lanjut ke proses berikutnya
    } catch (error) {
        return res
            .status(401)
            .json({
                success: false,
                message: "Token tidak valid atau sudah kedaluwarsa.",
            });
    }
};

// Fungsi satpam khusus untuk memastikan yang login adalah admin
const requireAdmin = (req, res, next) => {

        return res
            .status(403)
            .json({
                success: false,
                message: "Akses ditolak. Hanya admin yang diizinkan.",
            });
    }
    next();
};

        return res.status(403).json({

    }
    next();
};

const fs = require("fs");
const path = require("path");
const crypto = require("crypto");
const multer = require("multer");

// File disimpan di backend/uploads/koleksi dan diakses lewat /uploads/koleksi/<nama-file>
const UPLOAD_DIR = path.join(__dirname, "..", "uploads", "koleksi");
const PUBLIC_PREFIX = "/uploads/koleksi/";

// Samakan dengan MAX_IMAGE_MB di frontend (src/lib/validations/koleksi.ts)
const MAX_IMAGE_MB = 1;

const EXT_BY_MIME = {
    "image/jpeg": ".jpg",
    "image/png": ".png",
};

fs.mkdirSync(UPLOAD_DIR, { recursive: true });

const storage = multer.diskStorage({
    destination: (req, file, cb) => cb(null, UPLOAD_DIR),
    filename: (req, file, cb) => {
        // Nama dibuat sendiri (bukan dari nama file asli) agar aman dan tidak bentrok
        const unik = crypto.randomBytes(6).toString("hex");
        cb(null, `koleksi-${Date.now()}-${unik}${EXT_BY_MIME[file.mimetype]}`);
    },
});

const upload = multer({
    storage,
    limits: { fileSize: MAX_IMAGE_MB * 1024 * 1024, files: 1 },
    fileFilter: (req, file, cb) => {
        if (!EXT_BY_MIME[file.mimetype]) {
            const err = new Error("Hanya menerima file JPG atau PNG.");
            err.code = "INVALID_FILE_TYPE";
            return cb(err);
        }
        cb(null, true);
    },
}).single("gambar"); // nama field di FormData frontend

// Dipasang SETELAH authenticate/requireAdmin dan SEBELUM validate(),
// karena body berbentuk multipart sehingga baru terbaca setelah multer jalan.
const uploadGambar = (req, res, next) => {
    upload(req, res, (err) => {
        if (!err) return next();

        if (err instanceof multer.MulterError) {
            const message =
                err.code === "LIMIT_FILE_SIZE"
                    ? `Ukuran gambar maksimal ${MAX_IMAGE_MB} MB.`
                    : err.message;
            return res.status(400).json({ success: false, message });
        }
        if (err.code === "INVALID_FILE_TYPE") {
            return res
                .status(400)
                .json({ success: false, message: err.message });
        }
        return res
            .status(500)
            .json({ success: false, message: "Gagal mengunggah gambar." });
    });
};

// Hapus file lokal dari disk. URL luar (mis. Unsplash) atau path lain diabaikan.
const hapusFileGambar = (gambar) => {
    if (typeof gambar !== "string" || !gambar.startsWith(PUBLIC_PREFIX)) return;
    const namaFile = path.basename(gambar); // basename mencegah path traversal
    fs.unlink(path.join(UPLOAD_DIR, namaFile), () => {});
};

module.exports = { uploadGambar, hapusFileGambar, PUBLIC_PREFIX };

                "UPDATE verifikasi_kontak SET percobaan = percobaan + 1 WHERE kontak = ?",
                [kontak],
            );
            return false;
        }
        return true;
    },

        await db.query("DELETE FROM verifikasi_kontak WHERE kontak = ?", [

        error: "Tipe transaksi harus pemasukan atau pengeluaran",
    })
    // Aturan yang bergantung pada bank / e-wallet yang dipilih
    // (panjang rekening bank, awalan 08 untuk e-wallet).
    .superRefine((b, ctx) => {
        if (b.tipe !== "pengeluaran") return;

        const penyedia = cariPenyedia(b.jenisPencairan, b.penyedia);
        if (!penyedia) {
            ctx.addIssue({
                code: "custom",
                path: ["penyedia"],
                message: "Bank / e-wallet tujuan tidak valid",
            });
            return;
        }

        const pesan = pesanRekening(b.jenisPencairan, penyedia, b.noRekening);
        if (pesan) {
            ctx.addIssue({
                code: "custom",
                path: ["noRekening"],
                message: pesan,
            });
        }
    });

const { z } = require("zod");

// "" atau tidak diisi -> null (kolom di database boleh kosong)
const opsional = (inner) =>
    z.preprocess(
        (v) => (v == null || (typeof v === "string" && v.trim() === "") ? null : v),
        inner.nullable(),
    );

exports.pengunjungSchema = z
    .object({
        name: z.string().trim().min(3, "Nama minimal 3 karakter"),
        // 08xxxxxxxxxx (10-13 digit) atau format internasional +kodeNegara...
        phone: opsional(
            z
                .string()
                .trim()
                .regex(
                    /^(\+\d{8,15}|08\d{8,11})$/,
                    "Nomor telepon: 08xxxxxxxxxx (10-13 digit) atau +kodeNegara",
                ),
        ),
        email: opsional(
            z.string().trim().email("Format email salah").toLowerCase(),
        ),
        address: opsional(z.string().trim()),
    })
    .superRefine((d, ctx) => {
        if (!d.phone && !d.email) {
            ctx.addIssue({
                code: "custom",
                path: ["phone"],
                message: "Isi nomor telepon atau email (minimal salah satu)",
            });
        }
    });

"use client";

import { Plus } from "lucide-react";

import UserDetailModal from "@/components/user/UserDetailModal";
import UserFormModal from "@/components/user/UserFormModal";

import type { User } from "@/types/user";

export default function DataUserPage() {
    // Data dan logika (cari, tambah, ubah, hapus) ada di hook
    const { users, now, search, setSearch, addUser, editUser, removeUser } =
        useUsers();

    // State yang murni tampilan tetap di halaman
    const [isEditMode, setIsEditMode] = useState(false);
    const [isAddOpen, setIsAddOpen] = useState(false);
    const [detailUser, setDetailUser] = useState<User | null>(null);
    const [editTarget, setEditTarget] = useState<User | null>(null);
    const [deleteTarget, setDeleteTarget] = useState<User | null>(null);

    return (
        <div className="w-full animate-in fade-in duration-500 pb-10">
            <h1 className="text-[26px] font-bold text-black mb-8">Data User</h1>

                <SearchInput

                    placeholder="Cari Berdasarkan Nama / Email / No. Tlp"
                />

                {isEditMode && (
                    <button
                        onClick={() => setIsAddOpen(true)}
                        className="flex items-center gap-2 bg-[#222222] hover:bg-black text-white px-5 py-3 rounded-full transition-all font-semibold shadow-sm text-[14px] cursor-pointer animate-in zoom-in-95 duration-200">
                        <Plus className="w-5 h-5" />
                        Tambahkan User
                    </button>
                )}
            </div>

            <UserTable
                users={users}
                now={now}
                isEditMode={isEditMode}
                onToggleEditMode={() => setIsEditMode((prev) => !prev)}
                onDetail={setDetailUser}
                onEdit={setEditTarget}
                onDelete={setDeleteTarget}
            />

            {/* ================= MODALS ================= */}
            <UserDetailModal
                user={detailUser}
                now={now}
                onClose={() => setDetailUser(null)}
            />

            <ConfirmDeleteDialog
                open={!!deleteTarget}
                itemName={deleteTarget?.name ?? ""}
                onCancel={() => setDeleteTarget(null)}
                onConfirm={async () => {
                    if (deleteTarget) await removeUser(deleteTarget);
                    setDeleteTarget(null);
                }}
            />

            <UserFormModal
                open={!!editTarget}
                mode="edit"
                initialValues={
                    editTarget ? toEditValues(editTarget) : undefined
                }
                onClose={() => setEditTarget(null)}
                onSubmit={async (values) => {
                    if (editTarget) await editUser(editTarget, values);
                    setEditTarget(null);
                }}
            />

            <UserFormModal
                open={isAddOpen}
                mode="add"
                onClose={() => setIsAddOpen(false)}
                onSubmit={async (values) => {
                    await addUser(values);
                    setIsAddOpen(false);
                }}
            />
        </div>
    );
}

"use client";
import { ChevronLeft, ChevronRight } from "lucide-react";

interface Props {
    halaman: number; // halaman aktif (mulai dari 1)
    totalHalaman: number;
    totalData: number;
    perHalaman: number;
    onChange: (halaman: number) => void;
}

type ItemHalaman = number | "gap-awal" | "gap-akhir";

// Contoh total 12, aktif 6 -> 1 â€¦ 5 6 7 â€¦ 12
// Total <= 7 -> tampil semua: 1 2 3 4 5 6 7
const buatDaftarHalaman = (aktif: number, total: number): ItemHalaman[] => {
    if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);

    const hasil: ItemHalaman[] = [1];
    const awal = Math.max(2, aktif - 1);
    const akhir = Math.min(total - 1, aktif + 1);

    if (awal > 2) hasil.push("gap-awal");
    for (let i = awal; i <= akhir; i++) hasil.push(i);
    if (akhir < total - 1) hasil.push("gap-akhir");
    hasil.push(total);

    return hasil;
};

const TOMBOL_NAV =
    "flex items-center gap-1 h-9 px-3 rounded-full text-[13px] font-bold border-0 transition-colors cursor-pointer bg-gray-100 text-black hover:bg-gray-200 disabled:opacity-40 disabled:cursor-not-allowed disabled:hover:bg-gray-100";

export default function KeuanganPagination({
    halaman,
    totalHalaman,
    totalData,
    perHalaman,
    onChange,
}: Props) {
    if (totalData === 0) return null;

    const dari = (halaman - 1) * perHalaman + 1;
    const sampai = Math.min(halaman * perHalaman, totalData);

    return (
        <div className="px-6 py-4 border-t border-gray-100 flex flex-col sm:flex-row items-center justify-between gap-3">
            <span className="text-xs text-gray-500 font-medium">
                Menampilkan {dari}-{sampai} dari {totalData} transaksi
            </span>

            {totalHalaman > 1 && (
                <nav
                    aria-label="Navigasi halaman"
                    className="flex items-center gap-1.5 flex-wrap justify-center">
                    <button
                        type="button"
                        onClick={() => onChange(halaman - 1)}
                        disabled={halaman <= 1}
                        className={TOMBOL_NAV}>
                        <ChevronLeft className="w-4 h-4" />
                        <span className="hidden sm:inline">Sebelumnya</span>
                    </button>

                    {buatDaftarHalaman(halaman, totalHalaman).map((item) =>
                        typeof item === "number" ? (
                            <button
                                key={item}
                                type="button"
                                onClick={() => onChange(item)}
                                aria-current={
                                    item === halaman ? "page" : undefined
                                }
                                className={`w-9 h-9 rounded-full text-[13px] font-bold border-0 transition-colors cursor-pointer ${
                                    item === halaman
                                        ? "bg-[#222222] text-white"
                                        : "bg-gray-100 text-black hover:bg-gray-200"
                                }`}>
                                {item}
                            </button>
                        ) : (
                            <span
                                key={item}
                                className="w-6 text-center text-gray-400 font-bold select-none">
                                â€¦
                            </span>
                        ),
                    )}

                    <button
                        type="button"
                        onClick={() => onChange(halaman + 1)}
                        disabled={halaman >= totalHalaman}
                        className={TOMBOL_NAV}>
                        <span className="hidden sm:inline">Berikutnya</span>
                        <ChevronRight className="w-4 h-4" />
                    </button>
                </nav>
            )}
        </div>
    );
}

function Pil({ teks }: { teks: string | null }) {
    if (!teks) return <span className="text-gray-400">-</span>;
    return (
        <span className="px-2.5 py-1 rounded-full bg-gray-100 text-gray-800 border border-gray-200">
            {teks}
        </span>
    );
}

// Nama bank / e-wallet tujuan. Baris lama (belum ada nama) tampil "Transfer Bank" / "E-Wallet".
function PilPencairan({
    jenis,
    penyedia,
}: {
    jenis: string | null;
    penyedia: string | null;
}) {
    if (!jenis) return <span className="text-gray-400">-</span>;
    const bank = jenis === "Bank";
    return (
        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-gray-100 text-gray-900 border border-gray-200 font-bold">
            {bank ? (
                <Building2 className="w-3.5 h-3.5 text-blue-600" />
            ) : (
                <Smartphone className="w-3.5 h-3.5 text-emerald-600" />
            )}
            <span>{penyedia ?? (bank ? "Transfer Bank" : jenis)}</span>
        </span>
    );
}

function StatusPil({ status }: { status: string }) {
    const gaya = STATUS_STYLE[status] ?? {
        pil: "text-gray-600 bg-gray-100",

            <span className={`w-1.5 h-1.5 rounded-full ${gaya.titik}`}></span>
            {labelStatus(status)}
        </span>
    );
}

    const removeTimerRef = useRef<NodeJS.Timeout | null>(null);

        if (exitTimerRef.current) clearTimeout(exitTimerRef.current);
        if (removeTimerRef.current) clearTimeout(removeTimerRef.current);

        setToast((prev) => (prev ? { ...prev, isExiting: true } : null));

            setToast(null);
        }, 360);

                    prev ? { ...prev, isExiting: true } : null,

                    setToast(null);
                }, 360);
            }, 3000);

        [],
    );

    const IconComponent = currentStyle ? currentStyle.icon : CheckCircle2;

                            <IconComponent

"use client";
import {
    useCallback,
    useEffect,
    useLayoutEffect,
    useRef,
    useState,
} from "react";
import { type CacheKey, getCache, setCache } from "@/lib/cache";

interface UseCacheOptions {
    // Dipanggil bila pengambilan data gagal (mis. untuk menampilkan toast)
    onError?: (error: unknown) => void;
}

// Mengambil data dengan cache di localStorage (berlaku 5 menit).
// - Ada cache yang masih berlaku  -> langsung tampil, tanpa spinner dan tanpa request.
// - Belum ada / sudah kedaluwarsa -> ambil dari API, lalu simpan ke cache.
export function useCache<T>(
    key: CacheKey,
    fetcher: () => Promise<T>,
    { onError }: UseCacheOptions = {},
) {
    const [data, setData] = useState<T | null>(null);
    const [loading, setLoading] = useState(true);

    const dataRef = useRef<T | null>(null);
    const mountedRef = useRef(false);
    const fetcherRef = useRef(fetcher);
    const onErrorRef = useRef(onError);

    // Simpan fetcher & onError terbaru supaya refetch tidak ikut berubah tiap render
    useEffect(() => {
        fetcherRef.current = fetcher;
        onErrorRef.current = onError;
    });

    useEffect(() => {
        mountedRef.current = true;
        return () => {
            mountedRef.current = false;
        };
    }, []);

    // Ambil ulang dari API, lalu perbarui state dan cache
    const refetch = useCallback(async () => {
        setLoading(true);
        try {
            const result = await fetcherRef.current();
            setCache(key, result); // tetap diisi walau halaman sudah ditinggalkan
            if (mountedRef.current) {
                dataRef.current = result;
                setData(result);
            }
        } catch (error) {
            if (!mountedRef.current) return;
            if (onErrorRef.current) onErrorRef.current(error);
            else console.error(`Gagal memuat data "${key}":`, error);
        } finally {
            if (mountedRef.current) setLoading(false);
        }
    }, [key]);

    // Ubah data lokal (dan cache) tanpa memanggil API.
    // Bila belum ada data untuk diubah, ambil ulang dari API.
    const update = useCallback(
        (updater: (prev: T) => T) => {
            if (dataRef.current === null) {
                void refetch();
                return;
            }
            const next = updater(dataRef.current);
            dataRef.current = next;
            setData(next);
            setCache(key, next);
        },
        [key, refetch],
    );

    // Dibaca setelah komponen terpasang (bukan di useState) supaya hasil render
    // server dan browser sama. useLayoutEffect dipakai agar data dari cache sudah
    // tampil sebelum browser menggambar layar, jadi spinner tidak sempat berkedip.
    useLayoutEffect(() => {
        const cached = getCache<T>(key);
        if (cached !== null) {
            dataRef.current = cached;
            setData(cached);
            setLoading(false);
        } else {
            void refetch();
        }
    }, [key, refetch]);

    return { data, loading, refetch, update };
}

"use client";
import { type RefObject, useEffect } from "react";

export function useClickOutside<T extends HTMLElement>(
    ref: RefObject<T | null>,
    onOutside: () => void,
) {
    useEffect(() => {
        const handler = (e: MouseEvent) => {
            if (ref.current && !ref.current.contains(e.target as Node)) {
                onOutside();
            }
        };
        document.addEventListener("mousedown", handler);
        return () => document.removeEventListener("mousedown", handler);
    }, [ref, onOutside]);
}

export function useUsers() {
    const { notify } = useToast();

    // Data user: dari cache (5 menit) bila ada, kalau tidak dari API.
    // `update` mengubah tampilan sekaligus cache tanpa memuat ulang dari API.

        CACHE_KEYS.users,
        userService.getUsers,
    );
    const users = data ?? DAFTAR_KOSONG;

    // Waktu "sekarang" untuk teks "Offline 5 menit lalu"; ikut berdetak tiap 30 detik
    const [now, setNow] = useState(() => Date.now());

    // Status pengunjung berubah terus (cache 5 menit terlalu lama), jadi data diambil
    // ulang saat halaman dibuka lalu berkala selama halaman ini terbuka.
    useEffect(() => {
        void refetch();
        const timer = setInterval(() => {
            setNow(Date.now());
            void refetch();
        }, SEGARKAN_MS);
        return () => clearInterval(timer);
    }, [refetch]);

    const [search, setSearch] = useState("");

    // Filter pencarian: nama, email, atau nomor telepon (email / telepon bisa kosong)
    const filteredUsers = useMemo(() => {
        const q = search.toLowerCase();
        return users.filter(
            (u) =>
                u.name.toLowerCase().includes(q) ||
                (u.email ?? "").toLowerCase().includes(q) ||
                (u.phone ?? "").includes(search),
        );
    }, [users, search]);

    const addUser = async (values: UserFormValues) => {
        const created = await userService.createUser(values);
        update((prev) => [created, ...prev]);
        removeCache(CACHE_KEYS.dashboard); // jumlah user di dashboard bertambah
        notify({
            type: "create",
            title: "User ditambahkan",
            message: `${created.name} berhasil ditambahkan.`,
        });
    };

    const editUser = async (target: User, values: UserEditValues) => {
        const user: User = {
            ...target,
            name: values.name,
            phone: values.phone || null,
            email: values.email || null,
            address: values.address || null,
            createdAt: values.createdAt,
        };
        await userService.updateUser(user);
        update((prev) => prev.map((u) => (u.id === user.id ? user : u)));
        notify({
            type: "edit",
            title: "Data diperbarui",
            message: `Data ${user.name} berhasil diubah.`,
        });
    };

    const removeUser = async (user: User) => {
        await userService.deleteUser(user.id);
        update((prev) => prev.filter((u) => u.id !== user.id));
        removeCache(CACHE_KEYS.dashboard); // jumlah user di dashboard berkurang
        notify({
            type: "delete",
            title: "User dihapus",
            message: `${user.name} berhasil dihapus.`,
        });
    };

    return {
        users: filteredUsers,
        loading,
        now,
        search,
        setSearch,
        addUser,
        editUser,
        removeUser,
    };
}

// Cache data di localStorage dengan masa berlaku (TTL).
// Dipakai hook useCache supaya pindah menu tidak selalu memuat ulang dari API.

const PREFIX = "cache:v1:"; // naikkan versi bila bentuk data yang disimpan berubah
export const CACHE_TTL_MS = 5 * 60 * 1000; // 5 menit

// Satu kunci untuk setiap halaman
export const CACHE_KEYS = {
    dashboard: "dashboard",
    feedback: "feedback",
    keuangan: "keuangan",
    koleksi: "koleksi",
    users: "users",
} as const;
export type CacheKey = (typeof CACHE_KEYS)[keyof typeof CACHE_KEYS];

interface CacheEntry<T> {
    data: T;
    savedAt: number; // waktu disimpan (ms sejak epoch)
}

// Mengembalikan data cache, atau null bila belum ada / sudah kedaluwarsa / rusak.
export function getCache<T>(key: CacheKey): T | null {
    if (typeof window === "undefined") return null;
    try {
        const raw = localStorage.getItem(PREFIX + key);
        if (!raw) return null;

        const entry = JSON.parse(raw) as Partial<CacheEntry<T>> | null;
        if (!entry || typeof entry.savedAt !== "number" || entry.data == null) {
            localStorage.removeItem(PREFIX + key);
            return null;
        }

        const umur = Date.now() - entry.savedAt;
        if (umur < 0 || umur > CACHE_TTL_MS) {
            localStorage.removeItem(PREFIX + key);
            return null;
        }
        return entry.data;
    } catch {
        return null;
    }
}

export function setCache<T>(key: CacheKey, data: T): void {
    if (typeof window === "undefined") return;
    try {
        const entry: CacheEntry<T> = { data, savedAt: Date.now() };
        localStorage.setItem(PREFIX + key, JSON.stringify(entry));
    } catch {
        // Penyimpanan penuh / diblokir browser: abaikan, data tetap tampil dari state
    }
}

// Hapus cache tertentu (dipakai setelah tambah/ubah/hapus data)
export function removeCache(...keys: CacheKey[]): void {
    if (typeof window === "undefined") return;
    try {
        keys.forEach((key) => localStorage.removeItem(PREFIX + key));
    } catch {
        // abaikan
    }
}

// Hapus semua cache (dipakai saat login dan logout)
export function clearAllCache(): void {
    if (typeof window === "undefined") return;
    try {
        const daftar: string[] = [];
        for (let i = 0; i < localStorage.length; i++) {
            const k = localStorage.key(i);
            if (k && k.startsWith(PREFIX)) daftar.push(k);
        }
        daftar.forEach((k) => localStorage.removeItem(k));
    } catch {
        // abaikan
    }
}

import { api } from "@/lib/api";
import { DEFAULT_KOLEKSI_IMAGE } from "@/lib/constants";

// Mengubah nilai `gambar` dari database menjadi URL yang bisa dibuka browser.
// - "/uploads/koleksi/xxx.jpg"  -> "http://localhost:5000/uploads/koleksi/xxx.jpg"
// - URL penuh (http/https), blob:, dan data: dipakai apa adanya
// - kosong -> gambar bawaan
export function resolveImageUrl(src?: string | null): string {
    if (!src) return DEFAULT_KOLEKSI_IMAGE;
    if (/^(https?:|blob:|data:)/i.test(src)) return src;

    const base = (api.defaults.baseURL ?? "").replace(/\/$/, "");
    return `${base}${src.startsWith("/") ? "" : "/"}${src}`;
}

// Apakah tanggal (yyyy-mm-dd) masuk ke periode yang dipilih.
// hariIni juga yyyy-mm-dd; dibandingkan sebagai teks supaya tidak bergeser zona waktu.
export const cocokPeriode = (
    tanggal: string,
    periode: PeriodeKeuangan,
    hariIni: string,
) => {
    if (periode === "hari") return tanggal === hariIni;
    if (periode === "bulan") return tanggal.startsWith(hariIni.slice(0, 7));

// Mengembalikan pesan error, atau null kalau sudah sesuai.
export const pesanRekening = (
    jenis: JenisPencairan,
    penyedia: Penyedia,
    angka: string,
): string | null => {
    if (jenis === "Bank") {
        return angka.length === penyedia.digitMin
            ? null
            : `Nomor rekening ${penyedia.id} harus tepat ${penyedia.digitMin} digit angka`;
    }
    if (!angka.startsWith("08")) {
        return `Nomor HP ${penyedia.nama} harus diawali 08`;
    }
    return angka.length >= penyedia.digitMin &&
        angka.length <= penyedia.digitMax
        ? null
        : `Nomor HP ${penyedia.nama} harus ${penyedia.digitMin}-${penyedia.digitMax} digit`;
};

// Pembungkus localStorage yang aman untuk SSR dan JSON yang rusak
export function readStorage<T>(key: string, fallback: T): T {
    if (typeof window === "undefined") return fallback;
    try {
        const raw = localStorage.getItem(key);
        return raw ? (JSON.parse(raw) as T) : fallback;
    } catch {
        return fallback;
    }
}

export function writeStorage<T>(key: string, value: T): void {
    if (typeof window === "undefined") return;
    localStorage.setItem(key, JSON.stringify(value));
}

export const durasiLalu = (sejakIso: string, sekarang: number) => {
    const detik = Math.max(
        0,
        Math.floor((sekarang - new Date(sejakIso).getTime()) / 1000),
    );
    if (detik < 60) return "baru saja";
    const menit = Math.floor(detik / 60);
    if (menit < 60) return `${menit} menit lalu`;
    const jam = Math.floor(menit / 60);
    if (jam < 24) return `${jam} jam lalu`;
    return `${Math.floor(jam / 24)} hari lalu`;
};

import type { ZodError } from "zod";

export type FieldErrors<T> = Partial<Record<keyof T, string>>;

// Mengubah error zod menjadi { namaField: "pesan" } supaya mudah ditampilkan di bawah input
export function toFieldErrors<T>(error: ZodError): FieldErrors<T> {
    const result: Record<string, string> = {};
    for (const issue of error.issues) {
        const key = String(issue.path[0] ?? "");
        if (key && !result[key]) result[key] = issue.message; // ambil pesan pertama tiap field
    }
    return result as FieldErrors<T>;
}

// Pilihan dropdown. Ubah di sini kalau pilihannya perlu diganti.
export const JENIS_TIKET = ["Mandiri", "Guide"] as const;
export const JENIS_PENCAIRAN = ["Bank", "E-Wallet"] as const;

// Dibuat sebagai fungsi karena batas nominal bergantung pada saldo saat ini,
// sisa batas pencairan bulan ini, dan daftar bank / e-wallet dari backend
export const pencairanSchema = (
    saldo: number,
    sisaBatas: number,
    penyedia: DaftarPenyedia,
) =>
    z
        .object({
            nominal: z
                .string()
                .min(1, "Nominal wajib diisi")
                .refine(
                    (v) => parseNominal(v) > 0,
                    "Nominal harus lebih dari 0",
                )
                .refine(
                    (v) => parseNominal(v) <= saldo,
                    "Saldo tidak mencukupi",
                )
                .refine(
                    (v) => parseNominal(v) <= sisaBatas,
                    "Melebihi sisa batas pencairan bulan ini",
                ),
            jenisPencairan: z.enum(JENIS_PENCAIRAN),
            penyedia: z.string().min(1, "Pilih bank / e-wallet tujuan"),
            noRekening: z.string().min(1, "Nomor tujuan wajib diisi"),
        })
        .superRefine((v, ctx) => {
            const daftar =
                v.jenisPencairan === "Bank" ? penyedia.bank : penyedia.eWallet;
            const dipilih = daftar.find((p) => p.id === v.penyedia);
            if (!dipilih) {
                ctx.addIssue({
                    code: "custom",
                    path: ["penyedia"],
                    message: "Bank / e-wallet tidak valid",
                });
                return;
            }
            const pesan = pesanRekening(
                v.jenisPencairan,
                dipilih,
                v.noRekening.replace(/\D/g, ""),
            );
            if (pesan) {
                ctx.addIssue({
                    code: "custom",
                    path: ["noRekening"],
                    message: pesan,
                });
            }
        });

// Nominal tidak ada di sini: dihitung otomatis dari jumlah dan jenis tiket
