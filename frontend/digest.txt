"use client";
import { useState } from "react";
import Header from "@/components/layout/Header";
import Sidebar from "@/components/layout/Sidebar";
import NotificationModal from "@/components/layout/NotificationModal";
import { useAuthGuard } from "@/hooks/useAuthGuard";

export default function DashboardLayout({
    children,
}: {
    children: React.ReactNode;
}) {
    const [sidebarOpen, setSidebarOpen] = useState(false);
    const [notifOpen, setNotifOpen] = useState(false);
    const { checking, logout } = useAuthGuard();

    // Layar kosong saat mengecek status token login
    if (checking) return <div className="min-h-screen bg-[#f5f5f5]" />;

    return (
        <div className="min-h-screen flex bg-[#f5f5f5] text-[#080808] font-sans">
            <Sidebar
                open={sidebarOpen}
                onClose={() => setSidebarOpen(false)}
                onLogout={logout}
            />

            <div className="min-w-0 flex-1 flex flex-col h-screen overflow-y-auto">
                <Header
                    onMenuClick={() => setSidebarOpen(true)}
                    onNotificationClick={() => setNotifOpen(true)}
                />
                <main className="p-5 md:p-[40px] flex-1">{children}</main>
            </div>

            {notifOpen && (
                <NotificationModal onClose={() => setNotifOpen(false)} />
            )}
        </div>
    );
}

"use client";
import SearchInput from "@/components/ui/SearchInput";
import FeedbackTable from "@/components/feedback/FeedbackTable";
import FeedbackReplyModal from "@/components/feedback/FeedbackReplyModal";
import { useFeedback, STATUS_FILTERS } from "@/hooks/useFeedback";

export default function FeedbackPage() {
    const {
        feedbacks,
        totalFeedbacks,
        loading,
        submitting,
        searchQuery,
        setSearchQuery,
        statusFilter,
        setStatusFilter,
        modalReplyOpen,
        handleCloseModal,
        selectedFeedback,
        replyText,
        setReplyText,
        handleReplyClick,
        submitReply,
    } = useFeedback();

    return (
        <div className="w-full animate-in fade-in duration-500 pb-10 font-sans">
            <h1 className="text-[26px] font-bold text-black mb-8 mt-0">
                Feed Back
            </h1>

            {/* Search & filter status */}
            <div className="flex flex-col md:flex-row items-start md:items-center justify-between gap-4 mb-6">
                <SearchInput
                    value={searchQuery}
                    onChange={setSearchQuery}
                    placeholder="Cari Berdasarkan Nama / Ulasan"
                />

                <div className="flex items-center gap-2 bg-gray-100 p-1.5 rounded-full border border-gray-200">
                    {STATUS_FILTERS.map((filter) => (
                        <button
                            key={filter}
                            onClick={() => setStatusFilter(filter)}
                            className={`px-4 py-1.5 rounded-full text-xs font-bold transition-all cursor-pointer border-0 ${
                                statusFilter === filter
                                    ? "bg-black text-white shadow-sm"
                                    : "bg-transparent text-gray-600 hover:text-black"
                            }`}>
                            {filter}
                        </button>
                    ))}
                </div>
            </div>

            {loading && totalFeedbacks === 0 ? (
                <div className="flex justify-center items-center h-64">
                    <div className="animate-spin rounded-full h-10 w-10 border-4 border-[#222222] border-t-transparent" />
                </div>
            ) : (
                <FeedbackTable
                    feedbacks={feedbacks}
                    onReply={handleReplyClick}
                />
            )}

            <FeedbackReplyModal
                isOpen={modalReplyOpen}
                onClose={handleCloseModal}
                onSubmit={submitReply}
                feedback={selectedFeedback}
                replyText={replyText}
                setReplyText={setReplyText}
                loading={submitting}
            />
        </div>
    );
}

"use client";
import { useState } from "react";
import { ArrowRight } from "lucide-react";
import BaseModal from "@/components/ui/BaseModal";
import FieldError from "@/components/ui/FieldError";
import { formatRupiah } from "@/lib/format";
import { hitungNominal } from "@/lib/keuangan";
import { toFieldErrors, type FieldErrors } from "@/lib/validations";
import { isiSaldoSchema, JENIS_TIKET } from "@/lib/validations/keuangan";
import type { IsiSaldoInput, JenisTiket, Tarif } from "@/types/keuangan";

interface Props {
    isOpen: boolean;
    onClose: () => void;
    onSubmit: (data: IsiSaldoInput) => Promise<boolean>;
    submitting: boolean;
    tarif: Tarif;
}

interface FormState {
    nama: string;
    noTlp: string;
    tiket: string;
    jenis: string;
}

const FORM_AWAL: FormState = {
    nama: "",
    noTlp: "",
    tiket: "",
    jenis: JENIS_TIKET[0],
};

const LABEL = "block text-sm font-bold text-gray-700 mb-1.5";
const INPUT =
    "w-full px-4 py-3 rounded-xl border border-gray-300 outline-none font-semibold text-black focus:border-black focus:ring-1 focus:ring-black";

function IsiSaldoForm({
    onSubmit,
    submitting,
    tarif,
}: Omit<Props, "isOpen" | "onClose">) {
    const [form, setForm] = useState<FormState>(FORM_AWAL);
    const [errors, setErrors] = useState<FieldErrors<FormState>>({});

    const ubah = (field: keyof FormState, value: string) =>
        setForm((prev) => ({ ...prev, [field]: value }));

    const jumlahTiket = parseInt(form.tiket, 10) || 0;
    const jenisTiket = form.jenis as JenisTiket;
    const total = hitungNominal(jumlahTiket, jenisTiket, tarif);
    const rumus =
        jumlahTiket >= 1
            ? `${jumlahTiket} Ã— ${formatRupiah(tarif.hargaTiket)}${jenisTiket === "Guide" ? ` + pemandu ${formatRupiah(tarif.biayaPemandu)}` : ""}`
            : "Isi jumlah tiket";

    const handleSubmit = async (e: React.FormEvent) => {
        e.preventDefault();

        const hasil = isiSaldoSchema.safeParse(form);
        if (!hasil.success) {
            setErrors(toFieldErrors<FormState>(hasil.error));
            return;
        }
        setErrors({});

        await onSubmit({
            tipe: "pemasukan",
            nama: form.nama.trim(),
            noTelepon: form.noTlp,
            jumlahTiket,
            jenisTiket,
        });
    };

    return (
        <form
            onSubmit={handleSubmit}
            className="flex flex-col gap-4"
            noValidate>
            <div>
                <label className={LABEL}>Nama User</label>
                <input
                    type="text"
                    value={form.nama}
                    onChange={(e) => ubah("nama", e.target.value)}
                    placeholder="Nama Pengunjung"
                    className={INPUT}
                />
                <FieldError message={errors.nama} />
            </div>

            <div>
                <label className={LABEL}>Nomor Telepon</label>
                <input
                    type="tel"
                    inputMode="numeric"
                    value={form.noTlp}
                    onChange={(e) =>
                        ubah("noTlp", e.target.value.replace(/\D/g, ""))
                    }
                    maxLength={13}
                    placeholder="Contoh: 081234567890 (Hanya Angka)"
                    className={`${INPUT} font-mono`}
                />
                <FieldError message={errors.noTlp} />
            </div>

            <div className="flex gap-4">
                <div className="w-1/2">
                    <label className={LABEL}>Jumlah Tiket</label>
                    <input
                        type="number"
                        min="1"
                        value={form.tiket}
                        onChange={(e) => ubah("tiket", e.target.value)}
                        placeholder="1"
                        className={INPUT}
                    />
                    <FieldError message={errors.tiket} />
                </div>
                <div className="w-1/2">
                    <label className={LABEL}>Jenis Tiket</label>
                    <select
                        value={form.jenis}
                        onChange={(e) => ubah("jenis", e.target.value)}
                        className={`${INPUT} bg-white`}>
                        {JENIS_TIKET.map((j) => (
                            <option key={j} value={j}>
                                {j}
                            </option>
                        ))}
                    </select>
                    <FieldError message={errors.jenis} />
                </div>
            </div>

            {/* Di target ini kolom Nominal yang diketik; sekarang otomatis dari harga tiket */}
            <div className="rounded-xl bg-[#e6f3eb] px-4 py-3">
                <p className="text-xs font-semibold text-gray-500">
                    Total (bayar tunai)
                </p>
                <p className="text-xl font-bold text-black mt-0.5">
                    {formatRupiah(total)}
                </p>
                <p className="text-xs text-gray-500 mt-1">{rumus}</p>
            </div>

            <button
                type="submit"
                disabled={submitting || tarif.hargaTiket === 0}
                className="mt-4 w-full bg-[#222] hover:bg-black text-white font-bold py-3.5 rounded-xl flex items-center justify-center gap-2 cursor-pointer transition-colors shadow-sm disabled:opacity-50 border-0">
                {submitting ? (
                    "Memproses..."
                ) : (
                    <>
                        Konfirmasi Isi Saldo <ArrowRight className="w-5 h-5" />
                    </>
                )}
            </button>
        </form>
    );
}

    "w-full px-4 py-3 rounded-xl border border-gray-300 outline-none font-semibold text-black focus:border-black focus:ring-1 focus:ring-black";

function PencairanForm({
    onSubmit,
    submitting,
    saldo,
    batas,
}: Omit<Props, "isOpen" | "onClose">) {
    const [form, setForm] = useState<FormState>(FORM_AWAL);
    const [errors, setErrors] = useState<FieldErrors<FormState>>({});

    // maksimumPerBulan = 0 berarti data batas belum dimuat dari backend

    const tercapai = !belumDimuat && batas.sisa <= 0;
    const sekarang = new Date();
    const awalBulanDepan = new Date(
        sekarang.getFullYear(),
        sekarang.getMonth() + 1,
        1,
    );

    const ubah = (field: keyof FormState, value: string) =>
        setForm((prev) => ({ ...prev, [field]: value }));

        if (!hasil.success) {
            setErrors(toFieldErrors<FormState>(hasil.error));
            return;

            tipe: "pengeluaran",
            nama: ADMIN_NAME,

                <label className={LABEL}>Nominal (Rp)</label>
                <input
                    type="text"
                    inputMode="numeric"
                    value={form.nominal}
                    onChange={(e) =>
                        ubah("nominal", formatInputNominal(e.target.value))
                    }
                    placeholder="Contoh: 150.000"
                    className={INPUT}
                />
                <FieldError message={errors.nominal} />
            </div>

            <div>
                <label className={LABEL}>Nomor Rekening Tujuan</label>
                <input
                    type="text"
                    inputMode="numeric"
                    value={form.noRekening}
                    onChange={(e) =>
                        ubah("noRekening", e.target.value.replace(/\D/g, ""))
                    }
                    maxLength={16}
                    placeholder="Contoh: 1234567890 (Hanya Angka)"

                />

    "w-full px-4 py-3 rounded-xl border border-gray-300 outline-none font-semibold text-black focus:border-[#222222]";

export default function SejarahFormModal({
    isOpen,
    onClose,
    onSubmit,
    formData,
    setFormData,
    isEdit,
    loading,
}: Props) {
    const [imageError, setImageError] = useState("");

    // Bersihkan pesan error gambar setiap modal ditutup
    useEffect(() => {
        if (!isOpen) setImageError("");
    }, [isOpen]);

    const handleImageChange = (e: React.ChangeEvent<HTMLInputElement>) => {
        const file = e.target.files?.[0];
        if (!file) return;

        // Cek jenis (JPG/PNG) dan ukuran (maks MAX_IMAGE_MB) sebelum dipakai
        const error = validateImage(file);
        if (error) {
            setImageError(error);
            e.target.value = "";
            return;
        }

        setImageError("");
        if (formData.previewGambar) URL.revokeObjectURL(formData.previewGambar);
        setFormData({
            ...formData,
            fileGambar: file,
            previewGambar: URL.createObjectURL(file),
        });
    };

    // Pratinjau: file yang baru dipilih, jika tidak ada pakai gambar tersimpan
    const preview =
        formData.previewGambar ||
        (formData.gambar ? resolveImageUrl(formData.gambar) : "");

    return (
        <SejarahModalShell
            isOpen={isOpen}
            onClose={onClose}
            panelClassName="max-w-[500px] bg-white rounded-[24px] p-8 shadow-xl max-h-[90vh] overflow-y-auto">
            <button
                type="button"
                onClick={onClose}
                aria-label="Tutup"
                className="absolute top-6 right-6 z-10 border-0 bg-transparent cursor-pointer">
                <X className="w-6 h-6 text-gray-500 hover:text-black" />
            </button>

            <h2 className="text-[22px] font-bold text-black mb-6">
                {isEdit ? "Edit Koleksi Sejarah" : "Tambahkan Koleksi Baru"}
            </h2>

            <form onSubmit={onSubmit} className="flex flex-col gap-4">
                {/* MODE EDIT: pratinjau besar, hover untuk ganti foto */}
                {isEdit && (
                    <div className="mb-2">
                        <div className="relative w-full h-[200px] rounded-xl overflow-hidden bg-gray-100 group">
                            {preview && (
                                <Image
                                    src={preview}
                                    alt="Pratinjau koleksi"
                                    fill
                                    unoptimized
                                    className="object-cover"
                                />
                            )}
                            <div className="absolute inset-0 bg-black/50 opacity-0 group-hover:opacity-100 group-focus-within:opacity-100 flex items-center justify-center transition-opacity">
                                <label className="cursor-pointer bg-white text-black font-bold py-2 px-4 rounded-lg flex items-center gap-2 text-[13px] hover:bg-gray-100">
                                    <ImageIcon className="w-4 h-4" /> Ganti Foto
                                    <input
                                        type="file"
                                        accept=".jpg,.jpeg,.png"
                                        className="sr-only"
                                        onChange={handleImageChange}
                                    />
                                </label>
                            </div>
                        </div>
                        <p className="mt-1 text-xs text-gray-400 font-medium">
                            JPG atau PNG, maksimal {MAX_IMAGE_MB} MB
                        </p>
                        <FieldError message={imageError} />
                    </div>
                )}

                <div>
                    <label className={LABEL}>Nama Benda</label>
                    <input
                        type="text"
                        required
                        value={formData.nama || ""}
                        onChange={(e) =>
                            setFormData({ ...formData, nama: e.target.value })
                        }
                        placeholder="Contoh: Meriam Kuno"
                        className={INPUT}
                    />
                </div>

                {/* BARU: Jenis Benda dan Token */}
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                    <div>
                        <label className={LABEL}>Jenis Benda</label>
                        <input
                            type="text"
                            required
                            value={formData.jenis || ""}
                            onChange={(e) =>
                                setFormData({
                                    ...formData,
                                    jenis: e.target.value,
                                })
                            }
                            placeholder="Contoh: Senjata"
                            className={INPUT}
                        />
                    </div>
                    <div>
                        <label className={LABEL}>Token</label>
                        <input
                            type="text"
                            required
                            value={formData.token || ""}
                            onChange={(e) =>
                                setFormData({
                                    ...formData,
                                    token: e.target.value,
                                })
                            }
                            placeholder="Contoh: TKN-1234"
                            className={`${INPUT} font-mono`}
                        />
                    </div>
                </div>

                {/* MODE TAMBAH: tombol pilih file + thumbnail kecil */}
                {!isEdit && (
                    <div>
                        <label className={LABEL}>
                            Upload Gambar (.jpg / .png)
                        </label>
                        <div className="flex items-center gap-3">
                            <label className="cursor-pointer flex items-center justify-center gap-2 bg-gray-100 hover:bg-gray-200 text-black font-semibold py-3 px-4 rounded-xl border border-gray-300 w-full transition-colors">
                                <Upload className="w-4 h-4" /> Pilih File Gambar
                                <input
                                    type="file"
                                    accept=".jpg,.jpeg,.png"
                                    className="sr-only"
                                    onChange={handleImageChange}
                                    required={!formData.fileGambar}
                                />
                            </label>
                            {preview && (
                                <div className="relative w-12 h-12 shrink-0 rounded-lg overflow-hidden border border-gray-300">
                                    <Image
                                        src={preview}
                                        alt="Pratinjau"
                                        fill
                                        unoptimized
                                        className="object-cover"
                                    />
                                </div>
                            )}
                        </div>
                        <p className="mt-1 text-xs text-gray-400 font-medium">
                            JPG atau PNG, maksimal {MAX_IMAGE_MB} MB
                        </p>
                        <FieldError message={imageError} />
                    </div>
                )}

                <div>
                    <label className={LABEL}>Keterangan Penjelasan</label>
                    <textarea
                        required
                        rows={3}
                        value={formData.deskripsi || ""}
                        onChange={(e) =>
                            setFormData({
                                ...formData,
                                deskripsi: e.target.value,
                            })
                        }
                        placeholder="Deskripsi sejarah benda ini..."
                        className={`${INPUT} resize-none`}
                    />
                </div>

                <button
                    type="submit"
                    disabled={loading}
                    className={`mt-4 w-full text-white font-bold py-4 rounded-xl cursor-pointer transition-colors border-0 disabled:opacity-50 ${
                        isEdit
                            ? "bg-blue-600 hover:bg-blue-700"
                            : "bg-[#222222] hover:bg-black"
                    }`}>
                    {loading
                        ? "Menyimpan..."
                        : isEdit
                          ? "Konfirmasi Perubahan"
                          : "Konfirmasi Tambah"}
                </button>
            </form>
        </SejarahModalShell>
    );
}

"use client"; // Wajib ditambahkan untuk Next.js App Router

import React, {
    createContext,
    useContext,
    useState,
    useRef,
    useCallback,
} from "react";
import type { ToastOptions, ToastType } from "@/types/toast";
import {
    AlertCircle,
    AlertTriangle,
    CheckCircle2,
    Edit3,
    Info,
    Plus,
    Trash2,
    X,
    type LucideIcon,
} from "lucide-react";

interface ToastItem extends ToastOptions {
    id: number;
    isExiting: boolean;
}

interface ToastContextValue {
    notify: (options: ToastOptions) => void;
}

const ToastContext = createContext<ToastContextValue>({
    notify: () => {},
});

export const useToast = () => useContext(ToastContext);

const TOAST_STYLE: Record<
    ToastType,
    { icon: LucideIcon; iconColor: string; badge: string; bar: string }
> = {
    create: {
        icon: Plus,
        iconColor: "text-emerald-400",
        badge: "bg-emerald-500/15 border-emerald-500/30 text-emerald-400",
        bar: "bg-emerald-500",
    },
    edit: {

        badge: "bg-blue-500/15 border-blue-500/30 text-blue-400",

    },
    delete: {

        badge: "bg-rose-500/15 border-rose-500/30 text-rose-400",
        bar: "bg-rose-500",
    },
    success: {
        icon: CheckCircle2,
        iconColor: "text-emerald-400",
        badge: "bg-emerald-500/15 border-emerald-500/30 text-emerald-400",
        bar: "bg-emerald-500",
    },
    error: {
        icon: AlertCircle,
        iconColor: "text-red-400",
        badge: "bg-red-500/15 border-red-500/30 text-red-400",
        bar: "bg-red-500",
    },
    warning: {
        icon: AlertTriangle,
        iconColor: "text-yellow-400",
        badge: "bg-yellow-500/15 border-yellow-500/30 text-yellow-400",
        bar: "bg-yellow-500",
    },
    info: {
        icon: Info,
        iconColor: "text-blue-400",
        badge: "bg-blue-500/15 border-blue-500/30 text-blue-400",
        bar: "bg-blue-500",
    },
};

export function ToastProvider({ children }: { children: React.ReactNode }) {
    const [toast, setToast] = useState<ToastItem | null>(null);
    const exitTimerRef = useRef<NodeJS.Timeout | null>(null);
    const removeTimerRef = useRef<NodeJS.Timeout | null>(null);

    const closeToast = useCallback(() => {
        if (exitTimerRef.current) clearTimeout(exitTimerRef.current);
        if (removeTimerRef.current) clearTimeout(removeTimerRef.current);

        setToast((prev) => (prev ? { ...prev, isExiting: true } : null));

        removeTimerRef.current = setTimeout(() => {
            setToast(null);
        }, 360);
    }, []);

    const notify = useCallback(
        ({ type = "success", title, message }: ToastOptions) => {
            if (exitTimerRef.current) clearTimeout(exitTimerRef.current);
            if (removeTimerRef.current) clearTimeout(removeTimerRef.current);

            const newToast: ToastItem = {
                id: Date.now(),
                type,
                title,
                message,
                isExiting: false,
            };

            setToast(newToast);

            // Bertahan selama 3 detik di layar, kemudian mulai slide keluar ke kanan
            exitTimerRef.current = setTimeout(() => {
                setToast((prev) =>
                    prev ? { ...prev, isExiting: true } : null,
                );
                removeTimerRef.current = setTimeout(() => {
                    setToast(null);
                }, 360);
            }, 3000);
        },
        [],
    );

    const currentStyle = toast ? TOAST_STYLE[toast.type || "success"] : null;
    const IconComponent = currentStyle ? currentStyle.icon : CheckCircle2;

    return (
        <ToastContext.Provider value={{ notify }}>
            {children}
            {toast && currentStyle && (
                <div
                    key={toast.id}
                    className={`fixed top-6 right-6 z-[9999] w-full max-w-[380px] bg-[#1a1a1a] text-white rounded-[20px] shadow-2xl border border-white/10 overflow-hidden select-none pointer-events-auto ${
                        toast.isExiting ? "toast-slide-out" : "toast-slide-in"
                    }`}
                    style={{ willChange: "transform, opacity" }}>
                    <div className="p-4 flex items-start gap-3.5">
                        <div
                            className={`w-10 h-10 rounded-2xl flex items-center justify-center shrink-0 border ${currentStyle.badge}`}>
                            <IconComponent
                                className={`w-5 h-5 stroke-[2.5] ${currentStyle.iconColor}`}
                            />
                        </div>
                        <div className="flex-1 min-w-0 pt-0.5">
                            <h4 className="text-[14px] font-bold text-white tracking-tight leading-snug">
                                {toast.title}
                            </h4>
                            <p className="text-[12.5px] text-gray-300 mt-1 leading-relaxed">
                                {toast.message}
                            </p>
                        </div>
                        <button
                            onClick={closeToast}
                            className="p-1 rounded-full text-gray-400 hover:text-white hover:bg-white/10 transition-colors cursor-pointer border-0 bg-transparent shrink-0"
                            title="Tutup Notifikasi">
                            <X className="w-4 h-4" />
                        </button>
                    </div>
                    {!toast.isExiting && (
                        <div className="w-full bg-white/10 h-1 overflow-hidden">
                            <div
                                className={`h-full toast-progress-bar ${currentStyle.bar}`}
                            />
                        </div>
                    )}
                </div>
            )}
        </ToastContext.Provider>
    );
}

"use client";
import { type RefObject, useEffect } from "react";

export function useClickOutside<T extends HTMLElement>(
    ref: RefObject<T | null>,
    onOutside: () => void,
) {
    useEffect(() => {
        const handler = (e: MouseEvent) => {
            if (ref.current && !ref.current.contains(e.target as Node)) {
                onOutside();
            }
        };
        document.addEventListener("mousedown", handler);
        return () => document.removeEventListener("mousedown", handler);
    }, [ref, onOutside]);
}

export function useCountdown() {
    const [seconds, setSeconds] = useState(0);

    useEffect(() => {
        if (seconds <= 0) return;
        const timer = setTimeout(() => setSeconds((s) => s - 1), 1000);
        return () => clearTimeout(timer);
    }, [seconds]);

    return { seconds, start: setSeconds }; // start(60) untuk memulai

"use client";
import { useState, useEffect, useMemo } from "react";
import { useToast } from "@/context/ToastContext";
import { getErrorMessage } from "@/lib/api";
import * as feedbackService from "@/services/feedback.service";
import type { FeedbackData } from "@/types/feedback";

// Tetap diekspor dari sini supaya import lama
// (import { FeedbackData } from "@/hooks/useFeedback") tidak error.
export type { FeedbackData };

export const STATUS_FILTERS = [
    "Semua",
    "Belum Di Balas",
    "Telah Di Balas",
] as const;
export type StatusFilter = (typeof STATUS_FILTERS)[number];

export function useFeedback() {
    const { notify } = useToast();

    const [feedbacks, setFeedbacks] = useState<FeedbackData[]>([]);
    const [loading, setLoading] = useState(true);
    const [submitting, setSubmitting] = useState(false);
    const [searchQuery, setSearchQuery] = useState("");
    const [statusFilter, setStatusFilter] = useState<StatusFilter>("Semua");

    const [modalReplyOpen, setModalReplyOpen] = useState(false);
    const [selectedFeedback, setSelectedFeedback] =
        useState<FeedbackData | null>(null);
    const [replyText, setReplyText] = useState("");

    const fetchFeedbacks = async () => {
        setLoading(true);
        try {
            const data = await feedbackService.getFeedbacks();
            setFeedbacks(data);
        } catch (error) {
            notify({
                type: "error",
                title: "Gagal",
                message: getErrorMessage(error, "Gagal memuat data feedback"),
            });
        } finally {
            setLoading(false);
        }
    };

    useEffect(() => {
        fetchFeedbacks();
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, []);

    // Status diturunkan dari kolom balasan_admin (backend mengirimnya sebagai `balasan`)
    const filteredFeedbacks = useMemo(() => {
        const q = searchQuery.trim().toLowerCase();
        return feedbacks.filter((item) => {
            const matchesSearch =
                item.nama_user.toLowerCase().includes(q) ||
                item.komentar.toLowerCase().includes(q);

            const sudahDibalas = Boolean(item.balasan);
            const matchesStatus =
                statusFilter === "Semua" ||
                (statusFilter === "Telah Di Balas"
                    ? sudahDibalas
                    : !sudahDibalas);

            return matchesSearch && matchesStatus;
        });
    }, [feedbacks, searchQuery, statusFilter]);

    const handleReplyClick = (item: FeedbackData) => {
        setSelectedFeedback(item);
        setReplyText(item.balasan || ""); // jika sudah ada balasan, tampilkan di input
        setModalReplyOpen(true);
    };

    const handleCloseModal = () => {
        setModalReplyOpen(false);
        setReplyText("");
    };

    const submitReply = async (e: React.FormEvent) => {
        e.preventDefault();
        const teks = replyText.trim();
        if (!selectedFeedback || teks === "") return;

        const namaUser = selectedFeedback.nama_user;

        setSubmitting(true);
        try {
            await feedbackService.balasFeedback(selectedFeedback.id, teks);

            notify({
                type: "edit",
                title: "Balasan Berhasil Dikirim!",
                message: `Balasan untuk ulasan dari ${namaUser} telah berhasil dikirim.`,
            });
            handleCloseModal();
            fetchFeedbacks(); // ambil ulang dari database agar status ikut terbarui
        } catch (error) {
            notify({
                type: "error",
                title: "Gagal",
                message: getErrorMessage(
                    error,
                    "Terjadi kesalahan saat membalas.",
                ),
            });
        } finally {
            setSubmitting(false);
        }
    };

    return {
        feedbacks: filteredFeedbacks,
        totalFeedbacks: feedbacks.length,
        loading,
        submitting,
        searchQuery,
        setSearchQuery,
        statusFilter,
        setStatusFilter,
        modalReplyOpen,
        handleCloseModal,
        selectedFeedback,
        replyText,
        setReplyText,
        handleReplyClick,
        submitReply,
    };
}

    const [batas, setBatas] = useState<BatasPencairan>(BATAS_KOSONG);
    const [loading, setLoading] = useState(true);
    const [submitting, setSubmitting] = useState(false);

    const [tab, setTab] = useState<TabKeuangan>("masuk");
    const [rentang, setRentang] = useState<RentangTanggal>({
        dari: "",
        sampai: "",
    });
    const [searchQuery, setSearchQuery] = useState("");

    const [modalAktif, setModalAktif] = useState<ModalKeuangan>(null);
    const [detail, setDetail] = useState<TransaksiData | null>(null);

    const fetchKeuangan = async () => {
        setLoading(true);
        try {
            const res = await keuanganService.getKeuangan();
            setTransaksi(res.data);
            setSummary(res.summary);
            setTarif(res.tarif);
            setBatas(res.batasPencairan);
        } catch (error) {
            notify({
                type: "error",
                title: "Gagal",
                message: getErrorMessage(error, "Gagal memuat data keuangan"),
            });
        } finally {
            setLoading(false);
        }
    };

    useEffect(() => {
        fetchKeuangan();
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, []);

    // Filter tanggal + cari Id Trx. Kartu di atas TIDAK ikut filter (total keseluruhan);
    // angka di tab dan isi tabel ikut filter. Tanggal dibandingkan sebagai teks
    // yyyy-mm-dd, jadi batasnya inklusif dan tidak bergeser karena zona waktu.
    const tersaring = useMemo(() => {
        const q = searchQuery.trim();
        return transaksi.filter((item) => {
            if (rentang.dari && item.tanggal < rentang.dari) return false;
            if (rentang.sampai && item.tanggal > rentang.sampai) return false;
            if (q && !(item.idTrx ?? "").includes(q)) return false;
            return true;
        });
    }, [transaksi, rentang, searchQuery]);

    const jumlahMasuk = tersaring.filter((t) => t.tipe === "pemasukan").length;
    const jumlahKeluar = tersaring.length - jumlahMasuk;

    const daftar = useMemo(
        () =>
            tersaring.filter(
                (t) =>
                    t.tipe === (tab === "masuk" ? "pemasukan" : "pengeluaran"),
            ),
        [tersaring, tab],
    );

    // Kirim ke backend. Mengembalikan true kalau berhasil.
    // Saldo dicek lagi di backend, jadi aman walau data di layar sudah usang.
    const kirim = async (payload: TransaksiInput): Promise<boolean> => {
        setSubmitting(true);
        try {
            const hasil = await keuanganService.tambahKeuangan(payload);

            // Setelah berhasil, pindah ke tab yang relevan (sama seperti target)
            if (payload.tipe === "pemasukan") {
                notify({
                    type: "create",
                    title: "Transaksi Berhasil Ditambahkan!",
                    message: `Data transaksi untuk ${payload.nama} (${payload.jumlahTiket} tiket) senilai ${formatRupiah(hasil.jumlah)} berhasil dicatat.`,
                });
                setTab("masuk");
            } else {
                notify({
                    type: "success",
                    title: "Pencairan Berhasil!",
                    message: `Pencairan dana sebesar ${formatRupiah(hasil.jumlah)} berhasil diproses via ${payload.jenisPencairan}.`,
                });
                setTab("keluar");
            }

            setModalAktif(null);
            await fetchKeuangan();
            return true;
        } catch (error) {
            notify({
                type: "error",
                title: "Gagal",
                message: getErrorMessage(error, "Gagal memproses transaksi."),
            });
            return false;
        } finally {
            setSubmitting(false);
        }
    };

    const handleIsiSaldo = (data: IsiSaldoInput) => kirim(data);
    const handlePencairan = (data: PencairanInput) => kirim(data);

    return {
        daftar,
        jumlahMasuk,
        jumlahKeluar,
        summary,
        tarif,
        batas,
        loading,
        submitting,
        tab,
        setTab,
        rentang,
        setRentang,
        searchQuery,
        setSearchQuery,
        modalAktif,
        setModalAktif,
        detail,
        setDetail,
        handleIsiSaldo,
        handlePencairan,
    };
}

import { useEffect, useState } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { useToast } from "@/context/ToastContext";
import { getErrorMessage } from "@/lib/api";
import { saveSession } from "@/lib/auth";
import { loginSchema } from "@/lib/validations/auth";
import { type FieldErrors, toFieldErrors } from "@/lib/validations";
import * as authService from "@/services/auth.service";
import type { LoginPayload } from "@/types/auth";

export function useLogin() {
    const router = useRouter();
    const searchParams = useSearchParams();
    const { notify } = useToast();

    const [form, setForm] = useState<LoginPayload>({ email: "", password: "" });
    const [errors, setErrors] = useState<FieldErrors<LoginPayload>>({});
    const [loading, setLoading] = useState(false);

    useEffect(() => {
        if (searchParams.get("resetSuccess") === "true") {
            notify({
                type: "success",
                title: "Berhasil",
                message: "Sandi berhasil diperbarui!",
            });
            router.replace("/login");
        }
    }, [searchParams, notify, router]);

    const submit = async (e: React.FormEvent<HTMLFormElement>) => {
        e.preventDefault();

        const hasil = loginSchema.safeParse(form);
        if (!hasil.success) {
            setErrors(toFieldErrors<LoginPayload>(hasil.error));
            return;
        }

        setErrors({});
        setLoading(true);

        try {
            const res = await authService.login(hasil.data);
            const role = (res.user?.role ?? "").toLowerCase();

            if (role !== "admin") {
                notify({
                    type: "error",
                    title: "Akses ditolak",
                    message: "Akun ini bukan merupakan Admin.",
                });
                setLoading(false);
                return;
            }

            saveSession(res.token, role);
            notify({
                type: "success",
                title: "Berhasil",
                message: res.message || "Login berhasil",
            });

            // Biarkan loading tetap true selama jeda 1.5 detik menuju dashboard
            setTimeout(() => {
                router.replace("/dashboard");
            }, 1500);
        } catch (error) {
            notify({
                type: "error",
                title: "Login gagal",
                message: getErrorMessage(error, "Gagal melakukan login"),
            });
            setLoading(false);
        }
    };

    return { form, setForm, errors, loading, submit };
}

export function useResetRequest() {
    const router = useRouter();
    const { notify } = useToast();
    const { seconds, start } = useCountdown();

    const [email, setEmail] = useState("");
    const [code, setCode] = useState("");
    const [verifiedEmail, setVerifiedEmail] = useState("");
    const [errors, setErrors] = useState<
        FieldErrors<{ email: string; code: string }>
    >({});
    const [loading, setLoading] = useState(false);
    const [isExpired, setIsExpired] = useState(false);

    useEffect(() => {
        if (sessionStorage.getItem("reset_completed") === "true") {
            setIsExpired(true);
        }
    }, []);

    const sendCode = async () => {
        const hasil = forgotSchema.safeParse({ email });
        if (!hasil.success) {
            setErrors(toFieldErrors(hasil.error));
            return;
        }
        setErrors({});
        start(60);
        sessionStorage.removeItem("reset_completed");
        try {
            await authService.forgotPassword({ email: hasil.data.email });
            notify({
                type: "success",
                title: "Berhasil",
                message: "Kode berhasil dikirim ke sistem",
            });
            setVerifiedEmail(hasil.data.email);
        } catch (error) {
            notify({
                type: "error",
                title: "Gagal",
                message: getErrorMessage(error, "Email tidak terdaftar!"),
            });
            setVerifiedEmail("");
            start(0);
        }
    };

    const confirmCode = async (e: React.FormEvent) => {
        e.preventDefault();
        const hasil = kodeSchema.safeParse({
            email: verifiedEmail || email,
            code,
        });
        if (!hasil.success) {
            setErrors(toFieldErrors(hasil.error));
            return;
        }
        setErrors({});
        setLoading(true);
        try {
            await authService.verifyResetCode(hasil.data);
            sessionStorage.setItem("reset_email", hasil.data.email);
            sessionStorage.setItem("reset_code", hasil.data.code);
            notify({
                type: "success",
                title: "Berhasil",
                message: "Kode valid! Silakan buat sandi baru.",
            });
            setTimeout(() => router.push("/sandi-baru"), 1000);
        } catch (error) {
            notify({
                type: "error",
                title: "Kode salah",
                message: getErrorMessage(
                    error,
                    "Kode konfirmasi salah atau kedaluwarsa",
                ),
            });
            setLoading(false);
        }
    };

    return {
        email,
        setEmail,
        code,
        setCode,
        errors,
        loading,
        isExpired,
        seconds,
        sendCode,
        confirmCode,

"use client";

import { useEffect, useMemo, useState } from "react";
import { useToast } from "@/context/ToastContext";
import * as userService from "@/services/user.service";
import type { User, UserFormValues } from "@/types/user";

export function useUsers() {
    const { notify } = useToast();
    const [users, setUsers] = useState<User[]>([]);
    const [loading, setLoading] = useState(true);
    const [search, setSearch] = useState("");

    // Data dibaca setelah komponen tampil (bukan di dalam useState) supaya
    // hasil render di server dan di browser sama, tidak memicu hydration mismatch.
    useEffect(() => {
        let active = true;
        userService.getUsers().then((data) => {
            if (!active) return;
            setUsers(data);
            setLoading(false);
        });
        return () => {
            active = false;
        };
    }, []);

    // Filter pencarian: nama, email, atau nomor telepon
    const filteredUsers = useMemo(() => {
        const q = search.toLowerCase();
        return users.filter(
            (u) =>
                u.name.toLowerCase().includes(q) ||
                u.email.toLowerCase().includes(q) ||
                u.phone.includes(search),
        );
    }, [users, search]);

    const addUser = async (values: UserFormValues) => {
        const created = await userService.createUser(values);
        setUsers((prev) => [created, ...prev]);
        notify({
            type: "create",
            title: "User ditambahkan",
            message: `${created.name} berhasil ditambahkan.`,
        });
    };

    const editUser = async (user: User) => {
        await userService.updateUser(user);
        setUsers((prev) => prev.map((u) => (u.id === user.id ? user : u)));
        notify({
            type: "edit",
            title: "Data diperbarui",
            message: `Data ${user.name} berhasil diubah.`,
        });
    };

    const removeUser = async (user: User) => {
        await userService.deleteUser(user.id);
        setUsers((prev) => prev.filter((u) => u.id !== user.id));
        notify({
            type: "delete",
            title: "User dihapus",
            message: `${user.name} berhasil dihapus.`,
        });
    };

    return {
        users: filteredUsers,
        loading,
        search,
        setSearch,
        addUser,
        editUser,
        removeUser,
    };
}

import { z } from "zod";

// 1. Session Management
// ==========================================
const TOKEN_KEY = "adminToken";
const ROLE_KEY = "adminRole";

export function getToken(): string | null {
    if (typeof window === "undefined") return null;
    return localStorage.getItem(TOKEN_KEY);
}

export function saveSession(token: string, role?: string) {
    localStorage.setItem(TOKEN_KEY, token);
    if (role) localStorage.setItem(ROLE_KEY, role);
}

export function clearSession() {
    localStorage.removeItem(TOKEN_KEY);
    localStorage.removeItem(ROLE_KEY);
}

// ==========================================
// 2. Validation Schemas (Zod)
// ==========================================
export const emailSchema = z
    .string()
    .trim()
    .min(1, "Email wajib diisi")
    .email("Format email tidak valid")
    .toLowerCase(); // Tambahan opsional: memastikan email selalu format huruf kecil

export const loginSchema = z.object({
    email: emailSchema,
    password: z.string().min(1, "Password wajib diisi"),
});

export const forgotSchema = z.object({
    email: emailSchema,
});

export const kodeSchema = z.object({
    email: emailSchema,
    code: z.string().regex(/^\d{6}$/, "Kode harus 6 digit angka"),
});

    .object({
        newPassword: z

            .regex(/\d/, "Sandi harus mengandung angka"),
        confirmPassword: z.string().min(1, "Konfirmasi sandi wajib diisi"),
    })
    .refine((d) => d.newPassword === d.confirmPassword, {
        message: "Sandi dan konfirmasi sandi tidak cocok",
        path: ["confirmPassword"],
    });

// Type inference (opsional untuk membantu TypeScript di komponen React/Form)

};

// Tanggal hari ini (zona waktu perangkat) format ISO "yyyy-mm-dd"
export const todayISO = () => {
    const d = new Date();
    const pad = (n: number) => String(n).padStart(2, "0");
    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
};

const BULAN_SINGKAT = [
    "Jan",
    "Feb",
    "Mar",
    "Apr",
    "Mei",
    "Jun",
    "Jul",
    "Agu",
    "Sep",
    "Okt",
    "Nov",
    "Des",
];

// "2026-10-04" -> "04 Okt 2026"
// Dipecah manual (bukan new Date) supaya tanggal tidak bergeser karena zona waktu.
export const formatTanggalSingkat = (iso: string) => {
    if (!iso) return "";
    const [tahun, bulan, hari] = iso.split("-");
    const idx = parseInt(bulan, 10) - 1;
    if (!tahun || !hari || !(idx >= 0 && idx <= 11)) return iso;
    return `${hari.slice(0, 2)} ${BULAN_SINGKAT[idx]} ${tahun}`;
};

import { api } from "@/lib/api";
import { DEFAULT_KOLEKSI_IMAGE } from "@/lib/constants";

// Mengubah nilai `gambar` dari database menjadi URL yang bisa dibuka browser.
// - "/uploads/koleksi/xxx.jpg"  -> "http://localhost:5000/uploads/koleksi/xxx.jpg"
// - URL penuh (http/https), blob:, dan data: dipakai apa adanya
// - kosong -> gambar bawaan
export function resolveImageUrl(src?: string | null): string {
    if (!src) return DEFAULT_KOLEKSI_IMAGE;
    if (/^(https?:|blob:|data:)/i.test(src)) return src;

    const base = (api.defaults.baseURL ?? "").replace(/\/$/, "");
    return `${base}${src.startsWith("/") ? "" : "/"}${src}`;
}

// Pembungkus localStorage yang aman untuk SSR dan JSON yang rusak
export function readStorage<T>(key: string, fallback: T): T {
    if (typeof window === "undefined") return fallback;
    try {
        const raw = localStorage.getItem(key);
        return raw ? (JSON.parse(raw) as T) : fallback;
    } catch {
        return fallback;
    }
}

export function writeStorage<T>(key: string, value: T): void {
    if (typeof window === "undefined") return;
    localStorage.setItem(key, JSON.stringify(value));
}

import type { ZodError } from "zod";

export type FieldErrors<T> = Partial<Record<keyof T, string>>;

// Mengubah error zod menjadi { namaField: "pesan" } supaya mudah ditampilkan di bawah input
export function toFieldErrors<T>(error: ZodError): FieldErrors<T> {
    const result: Record<string, string> = {};
    for (const issue of error.issues) {
        const key = String(issue.path[0] ?? "");
        if (key && !result[key]) result[key] = issue.message; // ambil pesan pertama tiap field
    }
    return result as FieldErrors<T>;
}

};
