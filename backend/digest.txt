const crypto = require("crypto");

// Kode akses tiket: 10 karakter, pasti ada huruf besar, huruf kecil, dan angka
// (huruf/angka yang mirip seperti O/0 dan l/1 sengaja tidak dipakai).
const BESAR = "ABCDEFGHJKLMNPQRSTUVWXYZ";
const KECIL = "abcdefghijkmnopqrstuvwxyz";
const ANGKA = "23456789";
const SEMUA = BESAR + KECIL + ANGKA;

const ambil = (set) => set[crypto.randomInt(0, set.length)];

const buatKodeAkses = () => {
    const huruf = [ambil(BESAR), ambil(KECIL), ambil(ANGKA)];
    while (huruf.length < 10) huruf.push(ambil(SEMUA));
    // Acak urutan supaya 3 karakter pertama tidak selalu besar-kecil-angka
    for (let i = huruf.length - 1; i > 0; i--) {
        const j = crypto.randomInt(0, i + 1);
        [huruf[i], huruf[j]] = [huruf[j], huruf[i]];
    }
    return huruf.join("");
};

module.exports = { buatKodeAkses };

// Pengiriman notifikasi push lewat Firebase Cloud Messaging (FCM).
// Rahasia TIDAK ditulis di kode: letak file kunci akun layanan dibaca dari .env
//   FIREBASE_SERVICE_ACCOUNT=./firebase-service-account.json
// Jika belum diatur / paket firebase-admin belum dipasang, notifikasi dilewati (backend tetap jalan).

const path = require("path");

let messaging = null;

const ambilMessaging = () => {
    if (messaging) return messaging;

    const berkas = process.env.FIREBASE_SERVICE_ACCOUNT;
    if (!berkas) {
        console.warn("[PUSH] FIREBASE_SERVICE_ACCOUNT belum diatur di .env: notifikasi push dimatikan.");
        return null;
    }
    try {
        // Impor modular: berlaku untuk firebase-admin versi 10 ke atas (npm install firebase-admin)
        const { initializeApp, cert, getApps } = require("firebase-admin/app");
        const { getMessaging } = require("firebase-admin/messaging");
        const akun = require(path.resolve(berkas));
        const app = getApps().length > 0 ? getApps()[0] : initializeApp({ credential: cert(akun) });
        messaging = getMessaging(app);
        console.log("[PUSH] Firebase siap.");
    } catch (error) {
        // Tidak dicatat "sudah dicoba": setelah file / paket diperbaiki, balasan berikutnya langsung memakainya
        console.error("[PUSH] Firebase gagal dimulai:", error.message);
    }
    return messaging;
};

const TOKEN_MATI = new Set([
    "messaging/registration-token-not-registered",
    "messaging/invalid-registration-token",
]);

// tokens: daftar token FCM. Mengembalikan { terkirim, mati } (mati = token yang sudah tidak berlaku).
const kirimPush = async (tokens, { title, body, data }) => {
    const m = ambilMessaging();
    if (!m || tokens.length === 0) return { terkirim: 0, mati: [] };

    const hasil = await m.sendEachForMulticast({
        tokens,
        notification: { title, body },
        data: data ?? {},
        android: { priority: "high", notification: { channelId: "ulasan" } },
    });

    const mati = [];
    hasil.responses.forEach((r, i) => {
        if (!r.success) {
            console.error("[PUSH] Gagal ke satu HP:", r.error?.code, r.error?.message);
            if (TOKEN_MATI.has(r.error?.code)) mati.push(tokens[i]);
        }
    });
    console.log(`[PUSH] Terkirim ${hasil.successCount} dari ${tokens.length} HP.`);
    return { terkirim: hasil.successCount, mati };
};

module.exports = { kirimPush };

// Daftar bank & e-wallet tujuan pencairan (simulasi), satu sumber untuk backend dan web.
// Dikirim ke frontend lewat GET /api/keuangan, jadi frontend tidak perlu menyalin daftar ini.
// Ubah di sini kalau ada bank/e-wallet yang ditambah atau panjang rekeningnya berubah.

const bank = (id, nama, digit, contoh) => ({
    id,
    nama,
    digitMin: digit, // rekening bank: panjang harus tepat
    digitMax: digit,
    contoh,
});

const ewallet = (id, nama) => ({
    id,
    nama,
    digitMin: 10, // nomor HP: 10-13 digit
    digitMax: 13,
    contoh: "081234567890",
});

const DAFTAR_PENYEDIA = {
    bank: [
        bank("BCA", "Bank Central Asia (BCA)", 10, "1234567890"),
        bank("BRI", "Bank Rakyat Indonesia (BRI)", 15, "123401000123501"),
        bank("BNI", "Bank Negara Indonesia (BNI)", 10, "0123456789"),
        bank("Mandiri", "Bank Mandiri", 13, "1230004567890"),
        bank("BSI", "Bank Syariah Indonesia (BSI)", 10, "7123456789"),
        bank("CIMB", "CIMB Niaga", 13, "7012345678000"),
        bank("Danamon", "Bank Danamon", 10, "0036123456"),
        bank("Permata", "Bank Permata", 10, "4001234567"),
    ],
    eWallet: [
        ewallet("GoPay", "GoPay"),
        ewallet("OVO", "OVO"),
        ewallet("DANA", "DANA"),
        ewallet("ShopeePay", "ShopeePay"),
        ewallet("LinkAja", "LinkAja"),
    ],
};

// jenis = "Bank" | "E-Wallet"; mengembalikan null kalau id tidak ada di daftar
const cariPenyedia = (jenis, id) => {
    const daftar =
        jenis === "Bank" ? DAFTAR_PENYEDIA.bank : DAFTAR_PENYEDIA.eWallet;
    return daftar.find((p) => p.id === id) ?? null;
};

// Mengembalikan pesan error, atau null kalau nomor sudah sesuai aturan penyedia
const pesanRekening = (jenis, penyedia, angka) => {
    if (jenis === "Bank") {
        return angka.length === penyedia.digitMin
            ? null
            : `Nomor rekening ${penyedia.id} harus tepat ${penyedia.digitMin} digit angka`;
    }
    if (!angka.startsWith("08")) {
        return `Nomor HP ${penyedia.nama} harus diawali 08`;
    }
    return angka.length >= penyedia.digitMin &&
        angka.length <= penyedia.digitMax
        ? null
        : `Nomor HP ${penyedia.nama} harus ${penyedia.digitMin}-${penyedia.digitMax} digit`;
};

module.exports = { DAFTAR_PENYEDIA, cariPenyedia, pesanRekening };

// Harga tiket (simulasi). Ubah angka di sini kalau harga berubah.
// Nominal tiap transaksi disimpan di tabel keuangan, jadi transaksi lama
// tidak ikut berubah ketika harga diganti.

const hitungNominal = (jumlahTiket, jenisTiket) =>
    jumlahTiket * HARGA_TIKET + (jenisTiket === "Guide" ? BIAYA_PEMANDU : 0);

module.exports = { HARGA_TIKET, BIAYA_PEMANDU, hitungNominal };

const bcrypt = require("bcrypt");
const crypto = require("crypto"); // TAMBAHAN: Import modul crypto bawaan Node.js
const User = require("../models/user.model");
const { createToken } = require("../lib/jwt");

exports.login = async (req, res) => {
    try {
        const { email, password } = req.body;
        const user = await User.findByEmail(email);

        // REVISI: Pesan error disamakan agar tidak membocorkan apakah email terdaftar atau tidak
        if (!user)
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Email atau kata sandi salah",
                });

        const isValid = await bcrypt.compare(password, user.password);

        // REVISI: Pesan error disamakan
        if (!isValid)
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Email atau kata sandi salah",
                });

        const token = createToken(user);
        res.json({
            success: true,
            message: "Login berhasil",
            token,
            user: { email: user.email, role: user.role },
        });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.forgotPassword = async (req, res) => {
    try {
        const { email } = req.body;
        const user = await User.findByEmail(email);
        if (!user)
            return res
                .status(404)
                .json({ success: false, message: "Email tidak terdaftar" });

        // REVISI: Menggunakan crypto.randomInt untuk angka acak yang lebih aman
        const code = crypto.randomInt(100000, 999999).toString();

        await User.saveResetCode(email, code);

        // Simulasi pengiriman email (di proyek nyata gunakan Nodemailer)
        console.log(
            `[SIMULASI EMAIL] Kode reset untuk ${email} adalah: ${code}`,
        );

        res.json({
            success: true,
            message: "Kode reset telah dikirim (Cek terminal backend)",
        });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.verifyResetCode = async (req, res) => {
    try {
        const { email, code } = req.body;
        const user = await User.findByEmailAndResetCode(email, code);

        if (!user)
            return res.status(400).json({
                success: false,
                message: "Kode tidak valid atau kedaluwarsa",
            });
        res.json({ success: true, message: "Kode valid" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.resetPassword = async (req, res) => {
    try {
        const { email, code, newPassword } = req.body;
        const user = await User.findByEmailAndResetCode(email, code);

        if (!user)
            return res.status(400).json({
                success: false,
                message: "Kode tidak valid atau kedaluwarsa",
            });

        const hashed = await bcrypt.hash(newPassword, 10);
        await User.updatePassword(email, hashed);

        res.json({ success: true, message: "Kata sandi berhasil diperbarui" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

const Keuangan = require("../models/keuangan.model");
const {
    HARGA_TIKET,
    BIAYA_PEMANDU,
    hitungNominal,
} = require("../config/tarif");
const { DAFTAR_PENYEDIA } = require("../config/penyedia");
const { buatKodeAkses } = require("../config/akses");

// Isi Saldo lewat web admin = bayar tunai di loket (dibantu staff).
// Pembayaran non tunai nanti datang dari aplikasi mobile.
const PEMBAYARAN_WEB_ADMIN = "Tunai";

// Bentuk JSON yang dikirim ke web (Next.js) dan nanti ke mobile (Flutter)
const formatTransaksi = (item) => ({
    id: item.id,
    idTrx: item.id_trx,
    tanggal: item.tanggal.toLocaleDateString("sv-SE"), // YYYY-MM-DD (waktu lokal)
    deskripsi: item.keterangan,
    tipe: item.jenis.toLowerCase(), // 'Pemasukan' -> 'pemasukan'
    jumlah: Number(item.nominal),
    status: String(item.status || "Berhasil").toLowerCase(),
    nama: item.nama,
    noTelepon: item.no_telepon,
    jumlahTiket: item.jumlah_tiket,
    jenisTiket: item.jenis_tiket,
    jenisPembayaran: item.jenis_pembayaran,
    jenisPencairan: item.jenis_pencairan,
    penyedia: item.penyedia_pencairan, // nama bank / e-wallet tujuan
    noRekening: item.no_rekening,
    kodeAkses: item.kode_akses, // null = bukan transaksi tiket
    aksesKedaluwarsa: Boolean(item.akses_kedaluwarsa), // true = lewat 23:59
});

exports.getSemuaKeuangan = async (req, res) => {
    try {
        const [rows, summary, batasPencairan] = await Promise.all([
            Keuangan.getAll(),

        res.json({

            summary, // { totalPemasukan, totalPengeluaran, saldo }
            tarif: { hargaTiket: HARGA_TIKET, biayaPemandu: BIAYA_PEMANDU },
            batasPencairan, // { maksimumPerBulan, terpakaiBulanIni, sisa }
            penyedia: DAFTAR_PENYEDIA, // { bank: [...], eWallet: [...] }
        });
    } catch (error) {
        console.error("[KEUANGAN ERROR]", error);
        res.status(500).json({
            success: false,
            message: "Gagal mengambil data keuangan",
        });
    }
};

exports.tambahKeuangan = async (req, res) => {
    try {
        const b = req.body; // sudah divalidasi & dibersihkan oleh keuangan.schema
        let hasil;
        let nominal;
        let message;

        if (b.tipe === "pemasukan") {
            nominal = hitungNominal(b.jumlahTiket, b.jenisTiket);
            hasil = await Keuangan.create({
                jenis: "Pemasukan",
                nominal,
                keterangan: b.deskripsi || `Isi saldo - ${b.nama}`,
                nama: b.nama,
                noTelepon: b.noTelepon,
                jumlahTiket: b.jumlahTiket,
                jenisTiket: b.jenisTiket,
                jenisPembayaran: PEMBAYARAN_WEB_ADMIN,
                // Kode akses otomatis; admin memberikannya ke pengunjung untuk dipakai di aplikasi
                kodeAkses: buatKodeAkses(),
            });
            message = "Saldo berhasil ditambahkan";
        } else {
            nominal = b.jumlah;
            hasil = await Keuangan.createPencairan({
                jenis: "Pengeluaran",
                nominal,
                keterangan:
                    b.deskripsi ||
                    `Pencairan dana ke ${b.jenisPencairan} - ${b.penyedia}`,
                nama: b.nama,
                jenisPencairan: b.jenisPencairan,
                penyediaPencairan: b.penyedia,
                noRekening: b.noRekening,
            });
            message = "Pencairan berhasil diproses";
        }

        res.status(201).json({
            success: true,
            message,

const authenticate = (req, res, next) => {
    try {
        const authHeader = req.headers.authorization;
        if (!authHeader || !authHeader.startsWith("Bearer ")) {
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Akses ditolak. Token tidak ditemukan.",
                });

        const decoded = verifyToken(token);

        req.user = decoded; // Simpan data user (dari token) ke request
        next(); // Lanjut ke proses berikutnya
    } catch (error) {
        return res
            .status(401)
            .json({
                success: false,
                message: "Token tidak valid atau sudah kedaluwarsa.",
            });
    }
};

        try {

            return Array.isArray(arr) ? arr.filter((x) => typeof x === "string") : [];
        } catch {
            return [];
        }
    }
    return [raw];

const db = require("../config/db");
const { BATAS_PENCAIRAN_PER_BULAN } = require("../config/pencairan");

const NAMA_LOCK = "keuangan_pencairan";

const galat = (message, status) =>
    Object.assign(new Error(message), { status });

// Id Trx: 10 digit angka acak (boleh diawali 0), contoh: 0556277831.
// Dibuat di backend, bukan dari urutan id, jadi tidak bisa ditebak.
const buatIdTrx = () =>
    String(crypto.randomInt(0, 10_000_000_000)).padStart(10, "0");

const SQL_INSERT = `
    INSERT INTO keuangan
        (id_trx, jenis, nominal, keterangan, nama, no_telepon, jumlah_tiket,
         jenis_tiket, jenis_pembayaran, jenis_pencairan, penyedia_pencairan, no_rekening,
         status, pengunjung_id, kode_akses, kedaluwarsa_akses)
    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?,
            IF(? IS NULL, NULL, TIMESTAMP(CURDATE(), '23:59:59')))`;

// Simpan 1 baris. Kalau Id Trx kebetulan kembar (UNIQUE), buat ulang.
const insert = async (conn, d) => {
    for (let percobaan = 0; percobaan < 5; percobaan++) {
        const idTrx = buatIdTrx();
        try {
            const [result] = await conn.query(SQL_INSERT, [
                idTrx,
                d.jenis,
                d.nominal,
                d.keterangan,
                d.nama ?? null,
                d.noTelepon ?? null,
                d.jumlahTiket ?? null,
                d.jenisTiket ?? null,
                d.jenisPembayaran ?? null,
                d.jenisPencairan ?? null,
                d.penyediaPencairan ?? null,
                d.noRekening ?? null,
                d.status ?? "Berhasil",
                d.pengunjungId ?? null,
                d.kodeAkses ?? null,
                d.kodeAkses ?? null, // dipakai untuk menghitung kedaluwarsa (23:59:59 hari ini)
            ]);
            return { id: result.insertId, idTrx };
        } catch (err) {
            if (err.code !== "ER_DUP_ENTRY") throw err;
        }
    }
    throw new Error("Gagal membuat Id Trx yang unik");
};

// Saldo = SUM(pemasukan) - SUM(pengeluaran), hanya transaksi berstatus Berhasil.
// Transaksi Pending / Gagal belum / tidak dihitung.
const SQL_RINGKASAN = `
    SELECT
        COALESCE(SUM(CASE WHEN jenis = 'Pemasukan'   THEN nominal END), 0) AS pemasukan,
        COALESCE(SUM(CASE WHEN jenis = 'Pengeluaran' THEN nominal END), 0) AS pengeluaran
    FROM keuangan
    WHERE status = 'Berhasil'`;

// Total pencairan berhasil pada bulan kalender berjalan (memakai jam server database)
const SQL_PENCAIRAN_BULAN_INI = `

    FROM keuangan
    WHERE jenis = 'Pengeluaran'

      AND YEAR(tanggal) = YEAR(CURDATE())
      AND MONTH(tanggal) = MONTH(CURDATE())`;

const Keuangan = {

                 kedaluwarsa_akses = IF(kode_akses IS NULL, NULL, TIMESTAMP(CURDATE(), '23:59:59'))
             WHERE id = ? AND jenis = 'Pemasukan' AND status = 'Pending'`,

    // Pengguna membatalkan pesanan dari aplikasi: Pending -> Dibatalkan.
    // Hanya pesanan yang MASIH pending yang bisa dibatalkan (yang sudah dibayar tidak).
    batalkan: async (id) => {

            [id],
        );
        return result.affectedRows > 0;
    },

    // Kode akses sah: transaksi berhasil, belum lewat 23:59, milik akun ini
    // (atau kode dari loket / admin yang tidak terikat akun mana pun).

               AND kedaluwarsa_akses > NOW()

            saldo: totalPemasukan - totalPengeluaran,

        const terpakaiBulanIni = Number(row.total);
        return {
            maksimumPerBulan: BATAS_PENCAIRAN_PER_BULAN,
            terpakaiBulanIni,
            sisa: Math.max(0, BATAS_PENCAIRAN_PER_BULAN - terpakaiBulanIni),
        };
    },

    // Isi Saldo (pemasukan): saldo hanya bertambah, tidak perlu dikunci
    create: (data) => insert(db, data),
    createPesanan: (data) => insert(db, data), // pesanan dari aplikasi mobile

    // Pencairan (pengeluaran): saldo dicek dulu. Dikunci supaya dua pencairan
    // yang masuk bersamaan tidak sama-sama lolos dari saldo yang sama.
    createPencairan: async (data) => {
        const conn = await db.getConnection();
        try {
            const [[kunci]] = await conn.query("SELECT GET_LOCK(?, 5) AS ok", [
                NAMA_LOCK,
            ]);
            if (kunci.ok !== 1) {
                throw galat(
                    "Server sedang sibuk, coba lagi sebentar lagi",
                    503,
                );
            }

            const { saldo } = await Keuangan.getSummary(conn);
            if (data.nominal > saldo) {
                throw galat("Saldo tidak mencukupi untuk pencairan ini", 400);
            }

            const batas = await Keuangan.getBatasPencairan(conn);
            if (data.nominal > batas.sisa) {
                throw galat(
                    `Batas pencairan bulan ini terlampaui. Sisa batas: Rp ${batas.sisa.toLocaleString("id-ID")}`,
                    400,
                );
            }

            return await insert(conn, data);
        } finally {
            await conn
                .query("SELECT RELEASE_LOCK(?)", [NAMA_LOCK])
                .catch(() => {});
            conn.release();
        }
    },
};

        await db.query(
            `INSERT INTO verifikasi_kontak (kontak, kode, percobaan, kedaluwarsa)
             VALUES (?, ?, 0, DATE_ADD(NOW(), INTERVAL 10 MINUTE))
             ON DUPLICATE KEY UPDATE kode = ?, percobaan = 0,
                 kedaluwarsa = DATE_ADD(NOW(), INTERVAL 10 MINUTE)`,
            [kontak, kode, kode],
        );
    },

    // true jika kode benar, belum kedaluwarsa, dan belum kebanyakan salah
    cekKode: async (kontak, kode) => {
        const [rows] = await db.query(
            "SELECT kode, percobaan FROM verifikasi_kontak WHERE kontak = ? AND kedaluwarsa > NOW()",
            [kontak],
        );
        const row = rows[0];
        if (!row || row.percobaan >= MAKS_PERCOBAAN) return false;
        if (row.kode !== kode) {
            await db.query(
                "UPDATE verifikasi_kontak SET percobaan = percobaan + 1 WHERE kontak = ?",
                [kontak],
            );
            return false;
        }
        return true;
    },

    hapus: async (kontak) => {
        await db.query("DELETE FROM verifikasi_kontak WHERE kontak = ?", [kontak]);
    },
};

const { z } = require("zod");
const { cariPenyedia, pesanRekening } = require("../config/penyedia");

const hanyaAngka = (v) => v.replace(/\D/g, "");

const deskripsi = z
    .string()
    .trim()
    .max(255, "Deskripsi terlalu panjang")
    .optional();

// Isi Saldo -> dicatat sebagai Pemasukan.
// Nominal TIDAK dikirim klien: backend menghitungnya dari harga tiket.
const pemasukanSchema = z.object({
    tipe: z.literal("pemasukan"),
    nama: z
        .string()
        .trim()
        .min(3, "Nama minimal 3 karakter")
        .max(100, "Nama maksimal 100 karakter"),
    noTelepon: z
        .string()
        .trim()
        .optional()
        .refine(
            (v) => !v || /^08\d{8,11}$/.test(hanyaAngka(v)),
            "Nomor telepon diawali 08 (10-13 digit)",
        )
        .transform((v) => (v ? hanyaAngka(v) : undefined)),
    jumlahTiket: z
        .number({ error: "Jumlah tiket harus berupa angka" })
        .int("Jumlah tiket harus bilangan bulat")
        .min(1, "Jumlah tiket minimal 1"),
    jenisTiket: z.enum(["Mandiri", "Guide"], {
        error: "Jenis tiket tidak valid",
    }),
    deskripsi,
});

// Pencairan -> dicatat sebagai Pengeluaran
const pengeluaranSchema = z.object({
    tipe: z.literal("pengeluaran"),
    nama: z
        .string()
        .trim()
        .min(1, "Nama wajib diisi")
        .max(100, "Nama maksimal 100 karakter"),
    jumlah: z
        .number({ error: "Jumlah harus berupa angka" })
        .int("Jumlah harus bilangan bulat")
        .positive("Jumlah harus lebih dari 0"),
    jenisPencairan: z.enum(["Bank", "E-Wallet"], {
        error: "Jenis pencairan tidak valid",
    }),
    penyedia: z.string().trim().min(1, "Pilih bank / e-wallet tujuan"),
    noRekening: z
        .string()
        .trim()
        .refine(
            (v) => /^[\d\s-]+$/.test(v) && /^\d{8,16}$/.test(hanyaAngka(v)),
            "Nomor rekening harus 8-16 digit angka",
        )
        .transform(hanyaAngka),
    deskripsi,
});

exports.keuanganSchema = z
    .discriminatedUnion("tipe", [pemasukanSchema, pengeluaranSchema], {
        error: "Tipe transaksi harus pemasukan atau pengeluaran",
    })
    // Aturan yang bergantung pada bank / e-wallet yang dipilih
    // (panjang rekening bank, awalan 08 untuk e-wallet).
    .superRefine((b, ctx) => {
        if (b.tipe !== "pengeluaran") return;

        const penyedia = cariPenyedia(b.jenisPencairan, b.penyedia);
        if (!penyedia) {
            ctx.addIssue({
                code: "custom",
                path: ["penyedia"],
                message: "Bank / e-wallet tujuan tidak valid",
            });
            return;
        }

        const pesan = pesanRekening(b.jenisPencairan, penyedia, b.noRekening);
        if (pesan) {
            ctx.addIssue({
                code: "custom",
                path: ["noRekening"],
                message: pesan,
            });
        }
    });

    if (d.contact_type === "email") {
        if (!z.string().email().safeParse(d.contact).success) {
            ctx.addIssue({
                code: "custom",
                path: ["contact"],
                message: "Format email tidak valid",
            });
        }
    } else if (!/^\+\d{8,15}$/.test(d.contact)) {
        ctx.addIssue({
            code: "custom",
            path: ["contact"],
            message: "Format nomor telepon tidak valid (contoh +6281234567890)",
        });
    }
};

const kontakString = z.string().trim().min(1, "Kontak wajib diisi");

// contact-exists dan send-code
exports.kontakSchema = z
    .object({ contact_type: jenisKontak, contact: kontakString })
    .superRefine(cekFormatKontak)
    .transform((d) => ({ ...d, contact: rapikan(d.contact) }));

exports.verifySchema = z.object({
    contact: kontakString.transform(rapikan),
    code: z.string().regex(/^\d{6}$/, "Kode harus 6 digit angka"),
});

// Aturan password sama dengan aplikasi mobile (AuthValidators.passwordError)
const password = z
    .string()
    .min(10, "Password minimal 10 karakter")
    .max(50, "Password maksimal 50 karakter")
    .regex(/\p{L}/u, "Password harus mengandung huruf")
    .regex(/\d/, "Password harus mengandung angka")
    .regex(/[^\p{L}\p{N}\s]/u, "Password harus mengandung simbol");

exports.registerSchema = z
    .object({
        name: z.string().trim().min(3, "Nama minimal 3 karakter").max(50, "Nama maksimal 50 karakter"),
        contact_type: jenisKontak,
        contact: kontakString,
        password,
        code: z.string().regex(/^\d{6}$/, "Kode harus 6 digit angka"),
    })
    .superRefine(cekFormatKontak)
    .transform((d) => ({ ...d, contact: rapikan(d.contact) }));

// Login: email / nomor + password (kode verifikasi dikirim setelah password benar)
exports.loginSchema = z
    .object({
        contact_type: jenisKontak,
        contact: kontakString,

const { z } = require("zod");

// "" atau tidak diisi -> null (kolom di database boleh kosong)
const opsional = (inner) =>
    z.preprocess(
        (v) => (v == null || (typeof v === "string" && v.trim() === "") ? null : v),
        inner.nullable(),
    );

exports.pengunjungSchema = z
    .object({
        name: z.string().trim().min(3, "Nama minimal 3 karakter"),
        // 08xxxxxxxxxx (10-13 digit) atau format internasional +kodeNegara...
        phone: opsional(
            z
                .string()
                .trim()
                .regex(
                    /^(\+\d{8,15}|08\d{8,11})$/,
                    "Nomor telepon: 08xxxxxxxxxx (10-13 digit) atau +kodeNegara",
                ),
        ),
        email: opsional(
            z.string().trim().email("Format email salah").toLowerCase(),
        ),
        address: opsional(z.string().trim()),
    })
    .superRefine((d, ctx) => {
        if (!d.phone && !d.email) {
            ctx.addIssue({
                code: "custom",
                path: ["phone"],
                message: "Isi nomor telepon atau email (minimal salah satu)",
            });
        }
    });

const Feedback = require("../models/feedback.model");
const Perangkat = require("../models/perangkat.model");
const { kirimPush } = require("../config/firebase");

const POTONG = 120; // isi balasan di notifikasi dipotong supaya tidak kepanjangan

// Dipanggil setelah admin membalas ulasan -> notifikasi ke HP pemilik ulasan.
// Judul: "Ulasan"   Isi: "Ulasan di balas oleh admin : <isi balasan>"
const kirimNotifikasiBalasan = async (feedbackId, teksBalasan) => {
    const ulasan = await Feedback.findById(feedbackId);
    if (!ulasan?.pengunjung_id) return; // ulasan lama / tanpa akun mobile

    const tokens = await Perangkat.daftarToken(ulasan.pengunjung_id);
    if (tokens.length === 0) return;

    const ringkas = String(teksBalasan).trim().replace(/\s+/g, " ");
    const isi = ringkas.length > POTONG ? `${ringkas.slice(0, POTONG - 1)}â€¦` : ringkas;

    const { mati } = await kirimPush(tokens, {
        title: "Ulasan",
        body: `Ulasan di balas oleh admin : ${isi}`,
        data: { type: "review_reply", feedback_id: String(feedbackId) },
    });
    await Perangkat.hapusBanyak(mati); // buang token HP yang sudah tidak aktif
};

module.exports = { kirimNotifikasiBalasan };
