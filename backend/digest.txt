// Harga tiket (simulasi). Ubah angka di sini kalau harga berubah.
// Nominal tiap transaksi disimpan di tabel keuangan, jadi transaksi lama
// tidak ikut berubah ketika harga diganti.

const BIAYA_PEMANDU = 50000; // sekali per transaksi, hanya untuk jenis Guide

// 2 tiket Mandiri = 20.000 | 1 tiket Guide = 60.000 | 2 tiket Guide = 70.000
const hitungNominal = (jumlahTiket, jenisTiket) =>
    jumlahTiket * HARGA_TIKET + (jenisTiket === "Guide" ? BIAYA_PEMANDU : 0);

module.exports = { HARGA_TIKET, BIAYA_PEMANDU, hitungNominal };

const bcrypt = require("bcrypt");
const crypto = require("crypto"); // TAMBAHAN: Import modul crypto bawaan Node.js
const User = require("../models/user.model");
const { createToken } = require("../lib/jwt");

exports.login = async (req, res) => {
    try {
        const { email, password } = req.body;
        const user = await User.findByEmail(email);

        // REVISI: Pesan error disamakan agar tidak membocorkan apakah email terdaftar atau tidak
        if (!user)
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Email atau kata sandi salah",
                });

        const isValid = await bcrypt.compare(password, user.password);

        // REVISI: Pesan error disamakan
        if (!isValid)
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Email atau kata sandi salah",
                });

        const token = createToken(user);
        res.json({
            success: true,
            message: "Login berhasil",
            token,
            user: { email: user.email, role: user.role },
        });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.forgotPassword = async (req, res) => {
    try {
        const { email } = req.body;
        const user = await User.findByEmail(email);
        if (!user)
            return res
                .status(404)
                .json({ success: false, message: "Email tidak terdaftar" });

        // REVISI: Menggunakan crypto.randomInt untuk angka acak yang lebih aman
        const code = crypto.randomInt(100000, 999999).toString();

        await User.saveResetCode(email, code);

        // Simulasi pengiriman email (di proyek nyata gunakan Nodemailer)
        console.log(
            `[SIMULASI EMAIL] Kode reset untuk ${email} adalah: ${code}`,
        );

        res.json({
            success: true,
            message: "Kode reset telah dikirim (Cek terminal backend)",
        });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.verifyResetCode = async (req, res) => {
    try {
        const { email, code } = req.body;
        const user = await User.findByEmailAndResetCode(email, code);

        if (!user)
            return res.status(400).json({
                success: false,
                message: "Kode tidak valid atau kedaluwarsa",
            });
        res.json({ success: true, message: "Kode valid" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

exports.resetPassword = async (req, res) => {
    try {
        const { email, code, newPassword } = req.body;
        const user = await User.findByEmailAndResetCode(email, code);

        if (!user)
            return res.status(400).json({
                success: false,
                message: "Kode tidak valid atau kedaluwarsa",
            });

        const hashed = await bcrypt.hash(newPassword, 10);
        await User.updatePassword(email, hashed);

        res.json({ success: true, message: "Kata sandi berhasil diperbarui" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Terjadi kesalahan server",
        });
    }
};

const Keuangan = require("../models/keuangan.model");
const {
    HARGA_TIKET,
    BIAYA_PEMANDU,
    hitungNominal,
} = require("../config/tarif");

// Isi Saldo lewat web admin = bayar tunai di loket (dibantu staff).
// Pembayaran non tunai nanti datang dari aplikasi mobile.
const PEMBAYARAN_WEB_ADMIN = "Tunai";

// Bentuk JSON yang dikirim ke web (Next.js) dan nanti ke mobile (Flutter)
const formatTransaksi = (item) => ({
    id: item.id,
    idTrx: item.id_trx,
    tanggal: item.tanggal.toLocaleDateString("sv-SE"), // YYYY-MM-DD (waktu lokal)
    deskripsi: item.keterangan,
    tipe: item.jenis.toLowerCase(), // 'Pemasukan' -> 'pemasukan'
    jumlah: Number(item.nominal),
    status: String(item.status || "Berhasil").toLowerCase(),
    nama: item.nama,
    noTelepon: item.no_telepon,
    jumlahTiket: item.jumlah_tiket,
    jenisTiket: item.jenis_tiket,
    jenisPembayaran: item.jenis_pembayaran,
    jenisPencairan: item.jenis_pencairan,
    noRekening: item.no_rekening,
});

exports.getSemuaKeuangan = async (req, res) => {
    try {
        const [rows, summary, batasPencairan] = await Promise.all([
            Keuangan.getAll(),
            Keuangan.getSummary(),
            Keuangan.getBatasPencairan(),
        ]);

        res.json({
            success: true,
            data: rows.map(formatTransaksi),
            summary, // { totalPemasukan, totalPengeluaran, saldo }
            tarif: { hargaTiket: HARGA_TIKET, biayaPemandu: BIAYA_PEMANDU },
            batasPencairan, // { maksimumPerBulan, terpakaiBulanIni, sisa }
        });
    } catch (error) {
        console.error("[KEUANGAN ERROR]", error);
        res.status(500).json({
            success: false,
            message: "Gagal mengambil data keuangan",
        });
    }
};

exports.tambahKeuangan = async (req, res) => {
    try {
        const b = req.body; // sudah divalidasi & dibersihkan oleh keuangan.schema
        let hasil;
        let nominal;
        let message;

        if (b.tipe === "pemasukan") {
            nominal = hitungNominal(b.jumlahTiket, b.jenisTiket);
            hasil = await Keuangan.create({
                jenis: "Pemasukan",
                nominal,
                keterangan: b.deskripsi || `Isi saldo - ${b.nama}`,
                nama: b.nama,
                noTelepon: b.noTelepon,
                jumlahTiket: b.jumlahTiket,
                jenisTiket: b.jenisTiket,
                jenisPembayaran: PEMBAYARAN_WEB_ADMIN,
            });
            message = "Saldo berhasil ditambahkan";
        } else {
            nominal = b.jumlah;
            hasil = await Keuangan.createPencairan({
                jenis: "Pengeluaran",
                nominal,
                keterangan:
                    b.deskripsi || `Pencairan dana ke ${b.jenisPencairan}`,
                nama: b.nama,
                jenisPencairan: b.jenisPencairan,
                noRekening: b.noRekening,
            });
            message = "Pencairan berhasil diproses";
        }

        res.status(201).json({
            success: true,
            message,
            id: hasil.id,
            idTrx: hasil.idTrx,
            jumlah: nominal,
        });
    } catch (error) {
        // Galat yang sengaja dilempar model (saldo kurang, server sibuk)
        if (error.status) {
            return res
                .status(error.status)
                .json({ success: false, message: error.message });
        }
        console.error("[KEUANGAN ERROR]", error);
        res.status(500).json({
            success: false,
            message: "Gagal menyimpan transaksi keuangan",
        });
    }
};

const Koleksi = require("../models/koleksi.model");
const {
    hapusFileGambar,
    PUBLIC_PREFIX,
} = require("../middleware/upload.middleware");

// Dipakai jika koleksi dibuat tanpa gambar (sama dengan DEFAULT_KOLEKSI_IMAGE di frontend)
const DEFAULT_GAMBAR =
    "https://images.unsplash.com/photo-1544441893-675973e31985?w=600&auto=format&fit=crop&q=80";

// Path yang disimpan di database, contoh: /uploads/koleksi/koleksi-1730000000000-ab12cd34ef56.jpg
const pathGambar = (file) => (file ? `${PUBLIC_PREFIX}${file.filename}` : null);

const isTokenKembar = (error) => error && error.code === "ER_DUP_ENTRY";

exports.getSemuaKoleksi = async (req, res) => {
    try {
        const data = await Koleksi.getAll();
        res.json({ success: true, data });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Gagal mengambil data",
        });
    }
};

exports.tambahKoleksi = async (req, res) => {
    // Field "gambar" dari body sengaja diabaikan; yang dipakai hanya file hasil upload
    const gambar = pathGambar(req.file) || DEFAULT_GAMBAR;
    try {
        const insertId = await Koleksi.create({ ...req.body, gambar });
        res.status(201).json({
            success: true,
            message: "Data koleksi berhasil ditambahkan",
            id: insertId,
        });
    } catch (error) {
        hapusFileGambar(gambar); // jangan tinggalkan file yatim jika insert gagal
        if (isTokenKembar(error)) {
            return res.status(409).json({
                success: false,
                message: "Token sudah digunakan, gunakan token lain.",
            });
        }
        res.status(500).json({
            success: false,
            message: "Gagal menambah data, pastikan token unik!",
        });
    }
};

exports.ubahKoleksi = async (req, res) => {
    const gambarBaru = pathGambar(req.file);
    try {
        const lama = await Koleksi.getById(req.params.id);
        if (!lama) {
            hapusFileGambar(gambarBaru);
            return res.status(404).json({
                success: false,
                message: "Data koleksi tidak ditemukan",
            });
        }

        // Tidak upload gambar baru -> pertahankan gambar lama
        await Koleksi.update(req.params.id, {
            ...req.body,
            gambar: gambarBaru || lama.gambar,
        });

        if (gambarBaru) hapusFileGambar(lama.gambar); // buang file lama yang sudah tergantikan
        res.json({ success: true, message: "Data berhasil diperbarui" });
    } catch (error) {
        hapusFileGambar(gambarBaru);
        if (isTokenKembar(error)) {
            return res.status(409).json({
                success: false,
                message: "Token sudah digunakan, gunakan token lain.",
            });
        }
        res.status(500).json({
            success: false,
            message: "Gagal memperbarui data",
        });
    }
};

exports.hapusKoleksi = async (req, res) => {
    try {
        const lama = await Koleksi.getById(req.params.id);
        await Koleksi.delete(req.params.id);
        if (lama) hapusFileGambar(lama.gambar);
        res.json({ success: true, message: "Data berhasil dihapus" });
    } catch (error) {
        res.status(500).json({
            success: false,
            message: "Gagal menghapus data",
        });
    }
};

const { verifyToken } = require("../lib/jwt");

// Fungsi satpam untuk mengecek token
const authenticate = (req, res, next) => {
    try {
        const authHeader = req.headers.authorization;
        if (!authHeader || !authHeader.startsWith("Bearer ")) {
            return res
                .status(401)
                .json({
                    success: false,
                    message: "Akses ditolak. Token tidak ditemukan.",
                });
        }

        const token = authHeader.split(" ")[1];
        const decoded = verifyToken(token);

        req.user = decoded; // Simpan data user (dari token) ke request
        next(); // Lanjut ke proses berikutnya
    } catch (error) {
        return res
            .status(401)
            .json({
                success: false,
                message: "Token tidak valid atau sudah kedaluwarsa.",
            });
    }
};

// Fungsi satpam khusus untuk memastikan yang login adalah admin
const requireAdmin = (req, res, next) => {
    if (req.user.role !== "admin") {
        return res
            .status(403)
            .json({
                success: false,
                message: "Akses ditolak. Hanya admin yang diizinkan.",
            });
    }
    next();
};

module.exports = { authenticate, requireAdmin };

    destination: (req, file, cb) => cb(null, UPLOAD_DIR),
    filename: (req, file, cb) => {
        // Nama dibuat sendiri (bukan dari nama file asli) agar aman dan tidak bentrok
        const unik = crypto.randomBytes(6).toString("hex");
        cb(null, `koleksi-${Date.now()}-${unik}${EXT_BY_MIME[file.mimetype]}`);
    },
});

const upload = multer({
    storage,
    limits: { fileSize: MAX_IMAGE_MB * 1024 * 1024, files: 1 },
    fileFilter: (req, file, cb) => {
        if (!EXT_BY_MIME[file.mimetype]) {
            const err = new Error("Hanya menerima file JPG atau PNG.");
            err.code = "INVALID_FILE_TYPE";
            return cb(err);
        }
        cb(null, true);
    },
}).single("gambar"); // nama field di FormData frontend

// Dipasang SETELAH authenticate/requireAdmin dan SEBELUM validate(),
// karena body berbentuk multipart sehingga baru terbaca setelah multer jalan.
const uploadGambar = (req, res, next) => {
    upload(req, res, (err) => {
        if (!err) return next();

        if (err instanceof multer.MulterError) {
            const message =
                err.code === "LIMIT_FILE_SIZE"
                    ? `Ukuran gambar maksimal ${MAX_IMAGE_MB} MB.`
                    : err.message;
            return res.status(400).json({ success: false, message });
        }
        if (err.code === "INVALID_FILE_TYPE") {
            return res
                .status(400)
                .json({ success: false, message: err.message });
        }
        return res
            .status(500)
            .json({ success: false, message: "Gagal mengunggah gambar." });
    });
};

// Hapus file lokal dari disk. URL luar (mis. Unsplash) atau path lain diabaikan.
const hapusFileGambar = (gambar) => {
    if (typeof gambar !== "string" || !gambar.startsWith(PUBLIC_PREFIX)) return;
    const namaFile = path.basename(gambar); // basename mencegah path traversal
    fs.unlink(path.join(UPLOAD_DIR, namaFile), () => {});
};

module.exports = { uploadGambar, hapusFileGambar, PUBLIC_PREFIX };

    next();

const crypto = require("crypto");
const db = require("../config/db");
const { BATAS_PENCAIRAN_PER_BULAN } = require("../config/pencairan");

const NAMA_LOCK = "keuangan_pencairan";

const galat = (message, status) =>
    Object.assign(new Error(message), { status });

// Id Trx: 10 digit angka acak (boleh diawali 0), contoh: 0556277831.
// Dibuat di backend, bukan dari urutan id, jadi tidak bisa ditebak.
const buatIdTrx = () =>
    String(crypto.randomInt(0, 10_000_000_000)).padStart(10, "0");

const SQL_INSERT = `
    INSERT INTO keuangan
        (id_trx, jenis, nominal, keterangan, nama, no_telepon, jumlah_tiket,
         jenis_tiket, jenis_pembayaran, jenis_pencairan, no_rekening)
    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`;

// Simpan 1 baris. Kalau Id Trx kebetulan kembar (UNIQUE), buat ulang.
const insert = async (conn, d) => {
    for (let percobaan = 0; percobaan < 5; percobaan++) {
        const idTrx = buatIdTrx();
        try {
            const [result] = await conn.query(SQL_INSERT, [
                idTrx,
                d.jenis,
                d.nominal,
                d.keterangan,
                d.nama ?? null,
                d.noTelepon ?? null,
                d.jumlahTiket ?? null,
                d.jenisTiket ?? null,
                d.jenisPembayaran ?? null,
                d.jenisPencairan ?? null,
                d.noRekening ?? null,
            ]);
            return { id: result.insertId, idTrx };
        } catch (err) {
            if (err.code !== "ER_DUP_ENTRY") throw err;
        }
    }
    throw new Error("Gagal membuat Id Trx yang unik");
};

// Saldo = SUM(pemasukan) - SUM(pengeluaran), hanya transaksi berstatus Berhasil.
// Transaksi Pending / Gagal belum / tidak dihitung.
const SQL_RINGKASAN = `
    SELECT
        COALESCE(SUM(CASE WHEN jenis = 'Pemasukan'   THEN nominal END), 0) AS pemasukan,
        COALESCE(SUM(CASE WHEN jenis = 'Pengeluaran' THEN nominal END), 0) AS pengeluaran
    FROM keuangan
    WHERE status = 'Berhasil'`;

// Total pencairan berhasil pada bulan kalender berjalan (memakai jam server database)
const SQL_PENCAIRAN_BULAN_INI = `
    SELECT COALESCE(SUM(nominal), 0) AS total
    FROM keuangan
    WHERE jenis = 'Pengeluaran'
      AND status = 'Berhasil'
      AND YEAR(tanggal) = YEAR(CURDATE())
      AND MONTH(tanggal) = MONTH(CURDATE())`;

const Keuangan = {
    getAll: async () => {
        const [rows] = await db.query(
            "SELECT * FROM keuangan ORDER BY tanggal DESC, id DESC",
        );
        return rows;
    },

    getSummary: async (conn = db) => {
        const [[row]] = await conn.query(SQL_RINGKASAN);
        const totalPemasukan = Number(row.pemasukan);
        const totalPengeluaran = Number(row.pengeluaran);
        return {
            totalPemasukan,
            totalPengeluaran,
            saldo: totalPemasukan - totalPengeluaran,
        };
    },

    getBatasPencairan: async (conn = db) => {
        const [[row]] = await conn.query(SQL_PENCAIRAN_BULAN_INI);
        const terpakaiBulanIni = Number(row.total);
        return {
            maksimumPerBulan: BATAS_PENCAIRAN_PER_BULAN,
            terpakaiBulanIni,
            sisa: Math.max(0, BATAS_PENCAIRAN_PER_BULAN - terpakaiBulanIni),
        };
    },

    // Isi Saldo (pemasukan): saldo hanya bertambah, tidak perlu dikunci
    create: (data) => insert(db, data),

    // Pencairan (pengeluaran): saldo dicek dulu. Dikunci supaya dua pencairan
    // yang masuk bersamaan tidak sama-sama lolos dari saldo yang sama.
    createPencairan: async (data) => {
        const conn = await db.getConnection();
        try {
            const [[kunci]] = await conn.query("SELECT GET_LOCK(?, 5) AS ok", [
                NAMA_LOCK,
            ]);
            if (kunci.ok !== 1) {
                throw galat(
                    "Server sedang sibuk, coba lagi sebentar lagi",
                    503,
                );
            }

            const { saldo } = await Keuangan.getSummary(conn);
            if (data.nominal > saldo) {
                throw galat("Saldo tidak mencukupi untuk pencairan ini", 400);
            }

            const batas = await Keuangan.getBatasPencairan(conn);
            if (data.nominal > batas.sisa) {
                throw galat(
                    `Batas pencairan bulan ini terlampaui. Sisa batas: Rp ${batas.sisa.toLocaleString("id-ID")}`,
                    400,
                );
            }

            return await insert(conn, data);
        } finally {
            await conn
                .query("SELECT RELEASE_LOCK(?)", [NAMA_LOCK])
                .catch(() => {});
            conn.release();
        }
    },
};

module.exports = Keuangan;

const { z } = require("zod");

const hanyaAngka = (v) => v.replace(/\D/g, "");

const deskripsi = z
    .string()
    .trim()
    .max(255, "Deskripsi terlalu panjang")
    .optional();

// Isi Saldo -> dicatat sebagai Pemasukan.

const pemasukanSchema = z.object({
    tipe: z.literal("pemasukan"),
    nama: z
        .string()
        .trim()
        .min(3, "Nama minimal 3 karakter")
        .max(100, "Nama maksimal 100 karakter"),
    noTelepon: z
        .string()
        .trim()
        .optional()
        .refine(
            (v) => !v || /^08\d{8,11}$/.test(hanyaAngka(v)),
            "Nomor telepon diawali 08 (10-13 digit)",
        )
        .transform((v) => (v ? hanyaAngka(v) : undefined)),
    jumlahTiket: z
        .number({ error: "Jumlah tiket harus berupa angka" })
        .int("Jumlah tiket harus bilangan bulat")
        .min(1, "Jumlah tiket minimal 1"),
    jenisTiket: z.enum(["Mandiri", "Guide"], {
        error: "Jenis tiket tidak valid",
    }),
    deskripsi,
});

// Pencairan -> dicatat sebagai Pengeluaran
const pengeluaranSchema = z.object({
    tipe: z.literal("pengeluaran"),
    nama: z
        .string()
        .trim()
        .min(1, "Nama wajib diisi")
        .max(100, "Nama maksimal 100 karakter"),
    jumlah: z
        .number({ error: "Jumlah harus berupa angka" })
        .int("Jumlah harus bilangan bulat")
        .positive("Jumlah harus lebih dari 0"),
    jenisPencairan: z.enum(["Bank", "E-Wallet"], {
        error: "Jenis pencairan tidak valid",
    }),
    noRekening: z
        .string()
        .trim()
        .refine(
            (v) => /^[\d\s-]+$/.test(v) && /^\d{8,16}$/.test(hanyaAngka(v)),
            "Nomor rekening harus 8-16 digit angka",
        )
        .transform(hanyaAngka),
    deskripsi,
});

exports.keuanganSchema = z.discriminatedUnion(

    { error: "Tipe transaksi harus pemasukan atau pengeluaran" },
